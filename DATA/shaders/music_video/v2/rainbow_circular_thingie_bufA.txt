
vec3 a = vec3(0.5, 0.5, 0.5);
vec3 b = vec3(0.5, 0.5, 0.5);
vec3 c = vec3(1.0, 1.0, 1.0);
vec3 d = vec3(0.00, 0.33, 0.67);

// isquilez color mapper
vec3 colorMap(float t) {
	return (a + b * cos(2. * 3.141 * (c * t + d)));
}

// Working on this as an oriented basis function for CPPNs
float dv(vec2 uv, int am, float angle) {
    float r = length(uv);
    
    float th = atan(uv.y, uv.x);
    
    //float rin = 32.;
    //float rscale = (1.+rin)/rin - 1. / (rin*r + 1.);
    float rscale = tanh(3.*r*r);
    
    float value = 1.8 + rscale * 0.7*cos(float(am)*th + angle) - r;
    return tanh(value*value);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // Normalized pixel coordinates (from 0 to 1)
    vec2 uv = fragCoord/iResolution.xy;
    uv -= 0.5;
    uv.x *= iResolution.x/iResolution.y;

    float scale = 2.5;
    
    float value = 0.0;
    vec3 color = vec3(0.0);
    for (int i=0; i<3; i++) {
    	float v = dv(uv * scale, 1+i, iTime * (0.8 + 0.4*float(i)) * float(i%2-1));
        value += v;
        color[i] = v;
    }
    //fragColor = vec4(value);
    //fragColor = vec4(tanh(value));
    color = mix(color, colorMap(1.5*value + 0.3*iTime), 0.15);
    fragColor.xyz = color;
    //fragColor.xyz = colorMap(value);
}
