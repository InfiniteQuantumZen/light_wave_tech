
vec3 palette(float t) {
    vec3 a = vec3(0.5, 0.5, 0.5);
    vec3 b = vec3(0.5, 0.5, 0.5);
    vec3 c = vec3(1.0, 1.0, 1.0);
    vec3 d = vec3(0.2, 0.5, 0.85); // Tonos cian, magenta, verde y violeta
    return a + b * cos(6.28318 * (c * t + d));
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // 1. Normalización de coordenadas
    vec2 uv = (fragCoord * 2.0 - iResolution.xy) / min(iResolution.x, iResolution.y);
    vec2 uv0 = uv;

  
    uv = abs(uv);


    float gridScale = 3.0 + sin(iTime * 0.2) * 1.0; 
    vec2 gridUV = fract(uv * gridScale) - 0.5;


    float linesX = sin(gridUV.x * 20.0 + iTime * 2.0);
    float linesY = sin(gridUV.y * 20.0 - iTime * 2.0);
    

    float gridPattern = abs(linesX * linesY);

  
    float d = length(gridUV);
    d = sin(d * 7.0 + iTime * 0.5) / 8.0;
    d = abs(d);

    // Glow 
    float glow = pow(0.015 / (d + 0.001), 1.1);

    // 6. COLOR Y APAGADO CÍCLICO
    vec3 col = palette(length(uv0) + iTime * 0.15 + gridPattern * 0.2);


    vec3 finalColor = col * glow * (gridPattern * 1.5 + 0.2);
    float fadePulse = sin(iTime * 0.8) * 0.3 + 0.7;
    finalColor *= fadePulse;

 
    fragColor = vec4(pow(finalColor, vec3(0.9)), 1.0);
}
