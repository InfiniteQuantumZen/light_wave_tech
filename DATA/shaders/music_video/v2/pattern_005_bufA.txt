void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec2 r = iResolution.xy;
    vec2 uv = (fragCoord * 2.0 - r) / r.y;
    float t = iTime;

    vec4 o = vec4(0.0);

    for (float i = 0.0, z = 0.0, d = 0.0; z + i < 70.0; i += 1.0)
    {
        vec3 p = abs(z * normalize(vec3(uv, 1.0)));

        p.z += t * 5.0;

        p += sin(p * 1.5 + p.yzx);
       
        for (float j = 1.0; j <= 9.0; j += 1.0)
        {
            p += 0.4 * cos(round(0.23 * j * p) + 0.2 * t).zxy;
        }

        d = 0.1 * sqrt(length(p.xyy * p.yxy));
        z += d;

        vec3 glow = 0.5 + 0.5 * sin(vec3(0.0, 2.0, 4.0) + z * 0.15 + t * 0.25);
        o += vec4(glow * (1.0 + z * 0.05), 1.0) / max(d, 0.001);
    }

    fragColor = vec4(tanh(o.rgb / 3000.0), 1.0);
}
