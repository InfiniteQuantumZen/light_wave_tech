vec4 fire(vec2 u) {

    vec3 p;

    float i, d, s, n, T = iTime;

    vec4 o = vec4(0);

    for(; i++<4e1;
        d += s = .01 + abs(s)*.175,
        o += 1. / s,
        o += vec4(1,2,5,0) / length(u))
        for(p = vec3(u * d, d + T*3.),
            p += cos(p.z+T+p.yzx*.5)*.6,
            s = 2.-length(p.xy),
            p.xy *= mat2(cos(.5*T+vec4(0,33,11,0))),
                n = .016; n < .32; n += n )
                s += abs(dot(sin(T + .5*p/n ), p-p+4.)) * n;
    return (vec4(9,2,1,1) *o*o / d / 8e5);
}


void mainImage(out vec4 o, vec2 u) {
    float i,a,d,s,t=iTime;
    vec3  p = iResolution;    
    u = (u+u-p.xy)/p.y;
    //if (abs(u.x) > 1.4) { o*= i; return;}
    for(o*=i; i++<4e1; ) {
        p = vec3(u * d, d);
        s = 16.-length(p.xy);
        for (a = .01; a < 1.; a += a)
            p += cos(t-p.yzx*.5),
            p += cos(t-p.zxy*.3)*4.,
            s -= abs(dot(sin(t+t-.3*p.z+.5*p / a), vec3(a)));
        d += s = .07 + abs(s)*.15;
        o += 1./s + .08*vec4(9,2,1,0);
    }
    o = tanh((o + 3e2*fire(u)) / 4e2);
}
