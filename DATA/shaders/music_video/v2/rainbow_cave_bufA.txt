//Finally found a nice happy accident 

#define r(a) mat2(cos(a - vec4(0,11,33,0)))

void mainImage( out vec4 o, in vec2 F )
{
    vec2 R = iResolution.xy;
    o = vec4(0.);
    vec3 P, Q;
    float i = 0., g = i, d = 1., a, h = 0.4;
    
    for(; i < 99. && d > 1e-4; i++){
        o+=(1.+sin(3.*length(P.xy*3.)+g*0.+vec4(2,1,0,0)))/190.;
            
        g += d * .4;
        P = vec3((F - 0.5 * R) / R.y * g, g);
        P.z += iTime*2.;
        d = 2.-length(P.xy-vec2(sin(P.z*0.3),cos(P.z*0.3))*0.3);

        
        float s = 1.;
        for (a = 2.; a < 3e2; a += a) {
            Q = P * a;
            Q.xz *= r(a) ;
            d -= s*abs(dot(sin(Q-iTime*0.), Q-Q+1.)) / (a*1.2);
            s *= -0.75;
        }
    }
    
    //these two line give this the iridescent look
    o += 12. / i;
    o = 0.2/tanh(o*o)/vec4(3,2,1,0)*6./(g);/// pow(dot(F,F), -0.14);
}
