void mainImage(out vec4 o, vec2 u) {
    float i,s,d,j=.1*dot(sin(iTime+u), sin(u));
    vec3  c,r = iResolution,
          q,p,a = cos(vec3(2,3,1)+iTime/1e1);
    for(;i++ < 1e2; c +=vec3(5,3,1)+(1.+cos(.2*i+vec3(2,1,0)))/s)
        q = p = j+vec3((u+u - r.xy) / r.y*d, d),
        p = a*dot(a,p)-cross(a,p),
        q = abs(fract(p) - .5),
        d += s = .01+.5*max(abs(sin(iTime/4.+p.z*.1) * max(q.x,q.y)),
                     abs(.5+dot(sin(p/2.), ceil(cos(p.yzx/6.)))));
    o.rgb = tanh(c*c/2e6);
}
