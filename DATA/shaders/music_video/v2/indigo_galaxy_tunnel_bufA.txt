#define rot(a) mat2(cos(a+vec4(0.,11.,33.,0.)))

vec4 cmin(vec4 a, vec4 b){
if(a.w<b.w){return a;}

return b;
}

vec4 map(vec3 p){
p.z-= iTime;

vec3 op = p;

float id = floor(p.z)+floor(p.y)+floor(p.x);
float id2 = floor(p.z);

p.xy*=rot(id2);

p = fract(p)-0.5;

vec3 l = abs(p);

float e = 0.4;

float s = pow(pow(l.x,e)+pow(l.y,e)+pow(l.z,e),1./e) -0.5;


float st = 1.5;
float m = tanh(cos(op.z*0.1)*st)/tanh(st);

s= mix(1.-s,s,m*0.5+0.5);
s= abs(s)-0.1*(cos(length(op.xy)*1.));
s= s/20.+0.006;
vec3 col = mix(vec3(0.1,0.5,1.1),vec3(0.588,0.392,1.000),cos(op.z*10.));

return vec4(col,s);
}

vec3 trace(vec2 uv){

    vec3 ro = vec3(0.,0.,1.);
    vec3 rd = normalize(vec3(uv,-1.));
    
    vec3 acol = vec3(0.);
    
    vec3 p = ro;
    
    for(int i = 0; i<128;i++){
    vec4 d = abs(map(p));
    acol+= d.rgb/d.w;
    p+=rd*d.w;
   
    }
    
    return tanh(acol*0.00012);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // Normalized pixel coordinates (from 0 to 1)
    vec2 uv = (fragCoord-.5*iResolution.xy)/min(iResolution.x,iResolution.y);

    // Time varying pixel color
    vec3 col = trace(uv);
   
    // Output to screen
    fragColor = vec4(col,1.0);
}
