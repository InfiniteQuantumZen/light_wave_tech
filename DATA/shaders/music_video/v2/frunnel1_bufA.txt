#define T (iTime * 3.)
#define P(z) (vec3(tanh(cos((z) * .25) * .12) * 23., \
                   tanh(cos((z) * .13) * .15) * 14., (z)))
#define rot(a) mat2(cos(a), -sin(a), sin(a), cos(a))
#define R 2.25

float fractal(vec3 p) {
    p.xy -= P(p.z).xy;
    p.x -= 1.;
    p.xy = mod(p.xy+1., 8.) - 4.;
    p.z = mod(p.z, 2.) - 1.;
    vec4 q = vec4(p, 1.);
    for (float i = 0.; i < 8.; i++) {
        q.xyz  = abs(sin(q.xyz)) - 1.;
        q = q/pow(length(q.xyz),2.) -
            vec4(.0, .0, .0, 0.);
    } 
    return length(q.xyz)/q.w - .001;
}

void mainImage(out vec4 o, in vec2 u) {
    vec2 r = iResolution.xy; 
         u = (u - r.xy / 2.) / r.y;
         
    // It's safest to explicitly initialize p as well to prevent strict compiler warnings
    vec3 p = vec3(0.0),
         ro = P(T),
         la = P(T+2.);
         
    vec3 laz = normalize(la - ro),
         // Changed 0 to 0. to prevent strict type casting errors in desktop GLSL
         lax = cross(laz, vec3(0., -1., 0.)),
         lay = cross(lax, laz),
         rd = vec3(rot(sin(T*.13)*1.5)*u, 1.) * mat3(-lax, lay, laz);
         
    
    // FIX 1: Safely declare and initialize loop variables on separate lines
    float d = 0.0;
    float s = 0.0;
    float steps = 0.0;
    
    do {
        p = ro + rd * d;
        s = R - length(p.xy - P(p.z).xy);
        s = min(s, fractal(p));
        d += s;
        steps++;
    } while(d < 100. && s > .01 && steps < 70.);
    
    p = ro + rd * d;
    p.z = sin(p.z); 
    
    // FIX 2: Do all arithmetic in a temporary local variable, avoiding out-variable read-modify-writes
    vec3 col = abs(.1*p / dot(sin(p*4.),vec3(2./d)));
    col *= exp(-d/2.);
    
    // Finally, write to the output variable exactly once
    o = vec4(pow(col, vec3(.45)), 1.0);
}
