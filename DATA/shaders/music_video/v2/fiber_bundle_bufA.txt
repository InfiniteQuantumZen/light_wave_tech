void mainImage( out vec4 O, vec2 I ){
    float i, t, v, s;
    for (O*=i; i++<50.; t+=v*.1){
        // Ray position
        vec3 p = t*normalize(vec3(I+I,0)-iResolution.xyy);
        
        // Move back camera
        p.z += 38.;
        
        // Rotation & repetition in xz plane
        p.xz = mod(p.xz*mat2(cos(p.y*.02+iTime*.2-vec4(0,11,33,0))) -11.,22.)-11.;
        
        // Round vertically
        p.y = ceil(p.y);
        
        // Start with cylinder
        v = 5.*max(.03, length(p.xz)-5.);
        // Inside cylinder add some noise
        for (s=0.; s++<2.; v+=abs(dot(sin(p*s+iTime)/s, vec3(1))));
        
        // Color based on iter count
        O += (1.+cos(i*.2+vec4(5,1,3,0)))/v;
    }
    O = tanh(O/5e1);
}
