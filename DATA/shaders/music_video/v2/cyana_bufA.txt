void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec2 R = iResolution.xy;
    vec2 u = (fragCoord - 0.5 * R) / R.y;
    vec2 pos = u; 

   
    float t = iTime * 0.4; 
    
    
    float breath = (atan(iTime * 1.2) + atan(iTime * 0.7 + 1.0)) * 0.25 + 0.3;
    u *= 1.2 - breath * 0.555; 

    
    float distToCenter = length(u);
    float vortexAngle = exp(-distToCenter * 4.0) * sin(t * 1.5) * 1.2; 
    mat2 swirl = mat2(cos(vortexAngle), -sin(vortexAngle), sin(vortexAngle), cos(vortexAngle));
    u *= swirl; 

    vec3 finalColor = vec3(0.0);

    
    float ang = 8.4;
    mat2 rot = mat2(cos(ang), -sin(ang), sin(ang), cos(ang));
    
    
    vec3 colA = vec3(0.7, 0.0, 1.5);     
    vec3 colB = vec3(0.0, 1.2, 1.9);     
    vec3 colCore = vec3(2.0, 0.2, 1.0);  
    vec3 colDust = vec3(1.0, 0.8, 1.5);  

    const float ITERS = 65.0; 
    
    
    for (float i = 0.0; i < ITERS; i += 1.0) {
        
        
        vec2 flow = vec2(
            atan(u.y * 1.0 + t + i * 0.08) * 0.045,
            sin(u.x * 4.0 - t * 1.2 + i * 0.06) * 0.045
        );
        
        
        u += flow * (0.6 + distToCenter * 1.2); 

        u *= rot;
        
        
        u *= 1.05 + sin(i * 0.25 + t * 2.0) * 0.006; 

       
        vec2 p = u - vec2(0.5 + breath * 0.1, tan(t * 0.8 + i * 0.03) * 0.1);
        
        float d = length(vec2(p.x * 4.0, p.y * 0.2));

     
        float glow = 0.002 / (d * d + 0.001);
        glow *= (0.5 + breath * 0.7); 

       
        vec3 palette = mix(colA, colB, atan(i * 0.15 + t + length(u)) * 0.5 + 0.5);

      
        float core = smoothstep(0.15, 0.0, d);
        float intNucleo = clamp(core * (1.0 + breath * 1.5), 0.0, 1.0);
        palette = mix(palette, colCore, intNucleo);

        
        finalColor += glow * palette * exp2(-length(u) * 1.6);
    }

  
    float dust = 0.0;
    mat2 rotDust = mat2(cos(t), -sin(t), sin(t), cos(t));
    vec2 p_dust = pos * rotDust;

    for(float j = 1.0; j <= 3.0; j++) {
        
        vec2 p = fract(p_dust * (4.0 + j * 2.5) + vec2(sin(t * 0.6 + j), cos(t * 0.4 - j))) - 0.5;
        
        float twinkle = pow(sin(iTime * 5.0 + j * 5.0 + pos.x * 10.0) * 0.5 + 0.5, 2.0); 
        dust += (0.000012 / (dot(p, p) + 0.00001)) * twinkle;
    }
    
    
    finalColor += colDust * dust * (0.6 + breath * 0.8);

 
    finalColor *= 1.6; 
    
   
    finalColor = clamp((finalColor * (2.51 * finalColor + 0.03)) / (finalColor * (2.43 * finalColor + 0.59) + 0.14), 0.0, 1.0);
    
   
    finalColor = pow(finalColor, vec3(0.95));

   
    finalColor *= 1.0 - smoothstep(0.2, 1.7, length(pos));

    fragColor = vec4(finalColor, 1.0);
}
