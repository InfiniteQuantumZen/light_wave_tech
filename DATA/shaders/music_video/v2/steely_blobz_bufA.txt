#define T iTime
#define R(a) mat2(cos(a + vec4(0,33,11,0)))
#define N normalize

vec3 color;

// Commutative smooth minimum function. Provided by Tomkh, and taken 
// from Alex Evans's (aka Statix) talk: 
// http://media.lolrus.mediamolecule.com/AlexEvans_SIGGRAPH-2015.pdf
// Credited to Dave Smith @media molecule.
float smin(float a, float b, float k){

   float f = max(0., 1. - abs(b - a)/k);
   return min(a, b) - k*.25*f*f;
}

float box(vec3 p, float i) {
    p = abs(fract(p/i)*i - i/2.) - i*.05;
    return min(p.x, min(p.y, p.z));
}

float boxen(vec3 p) {
    float d = -9e9, i = 1e1;
    for(; i > .15; i *= .201)
        p.xz *= R(i),
        d = max(d, box(p, i));
    return d;
}

float blobs(vec3 p) {

    float s1, s2;

    // s1
    p.z -=  4.5;
    p.xy += .5;
    p.xz *= R(T/1e1);
    p += .15*(cos(.3*T+dot(cos(.4*T+p+cos(p)), p) *  p/2. ));
    s1 = length(p+.75) - 2.;

    // s2
    p.xz *= R(T/2e1);
    p -= .2*(cos(.2*T+dot(cos(.2*T+p+cos(p)), p) *  p/2. ));
    s2 = length(p-.75) - 2.;

    color += mix(vec3(6,2,1)*s1, vec3(1,2,6)*s2, smoothstep(.2, 1., s1-s2));

    // smin(s1, s2, x)
    return smin(s1, s2, .8);
}

bool hit;
float map(vec3 p) {
    float s = blobs(p),
          b = boxen(p);
    hit = s < b;
    return min(s, b);
}

// @iq
float AO(in vec3 pos, in vec3 nor) {
	float sca = 2.0, occ = 0.0;
    for( int i=0; i<5; i++ ){
    
        float hr = 0.01 + float(i)*0.5/4.0;        
        float dd = map(nor * hr + pos);
        occ += (hr - dd)*sca;
        sca *= 0.7;
    }
    return clamp( 1.0 - occ, 0.0, 1.0 );    
}

void mainImage(out vec4 o, in vec2 u) {
    float s, i;

    vec3 r = iResolution;
    u = (u+u - r.xy) / r.y;
    u.yx -= .3;

    vec3 p,D = N(vec3(u, 1));

    for(o *= i; i++ < 64.; )
        p += D * s * .8,
        o += s = map(p);
    
    // normal
    // tetrahedron technique: https://iquilezles.org/articles/normalsSDF/
    const float h = 0.005;
    const vec2 k = vec2(1,-1);
    r = N(k.xyy*map( p + k.xyy*h ) + 
          k.yyx*map( p + k.yyx*h ) + 
          k.yxy*map( p + k.yxy*h ) + 
          k.xxx*map( p + k.xxx*h ) );

    // reflection march
    if(hit) {
        vec3 ref;
        for(p += r*.05, D = reflect(D, r), s=i=0.; i++<32.; )
            p += D*s,
            s = map(p)*.8,
            ref += s;

        o.rgb += color/3e1*ref/3.;
    }

    o *= AO(p, r);
    o = tanh(o/1e1+.1*dot(u,u));
}
