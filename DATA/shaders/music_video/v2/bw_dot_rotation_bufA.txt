#define SCALE 50.0

// negative speed = moving backwards 
#define SPEED 60.0

#define SCENE_ROTATION_SPEED 0.2

#define PLANE_ROTATION_SPEED -1.0

#define DOT_SHARPNESS 5.0

mat2 rotate2d(float angle){
    return mat2(
        cos(angle), -sin(angle),
        sin(angle), cos(angle)
    );
}


float getColor(vec2 pixel, vec2 trans, float delay) {
    float value = cos(trans.x);
    value += cos(trans.y - (iTime - delay) * SPEED);
    value = (value + 2.0) / 4.0;
    value = pow(value, DOT_SHARPNESS);
    value *= abs(pixel.y * 2.0);
    return value;
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 pixel = (fragCoord -.5 * iResolution.xy) / iResolution.y;
    pixel *= rotate2d(iTime * SCENE_ROTATION_SPEED);
    vec2 trans = vec2(pixel.x / pixel.y, log(abs(pixel.y)));
    trans *= rotate2d(iTime * PLANE_ROTATION_SPEED);    
    trans *= SCALE;
    
    
    float value = cos(trans.x);
    value += cos(trans.y - iTime * SPEED);
    value = (value + 2.0) / 4.0;
    value = pow(value, DOT_SHARPNESS);
    // apply air perspective
    value *= abs(pixel.y * 2.0);
    fragColor = vec4(
        getColor(pixel, trans, sin(iTime * 1.0) * 0.015),
        getColor(pixel, trans, sin(iTime * 1.1) * 0.015),
        getColor(pixel, trans, sin(iTime * 1.2) * 0.015),
        1.0
    );
}
