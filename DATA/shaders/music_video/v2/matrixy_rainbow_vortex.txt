float TK = 1.;
float PI = 3.1415926535;

vec2 rot(vec2 p,float r){
	mat2 m = mat2(cos(r),sin(r),-sin(r),cos(r));
	return m*p;
}

vec2 pmod(vec2 p,float n){
	float np = 2.0*PI/n;
	float r = atan(p.x,p.y)-0.5*np;
	r = mod(r,np)-0.5*np;
	return length(p)*vec2(cos(r),sin(r));
}

float cube(vec3 p,vec3 s){
	vec3 q = abs(p);
	vec3 m = max(s-q,0.0);
	return length(max(q-s,0.0))-min(min(m.x,m.y),m.z);
}

float dist(vec3 p){
	p.z -= 1.*TK*iTime;
	p.xy = rot(p.xy,1.0*p.z);
	p.xy = pmod(p.xy,6.0);
	float k = 0.7;
	float zid = floor(p.z*k);
	p = mod(p,k)-0.5*k;
	for(int i = 0;i<4;i++){
		p = abs(p)-0.3;

		p.xy = rot(p.xy,1.0+zid+0.1*TK*iTime);
		p.xz = rot(p.xz,1.0+4.7*zid+0.3*TK*iTime);
	}
	return min(cube(p,vec3(0.3)),length(p)-0.4);
}



// 1. Put this function ABOVE your mainImage function
float rand(float seed) {
    // This is the classic GLSL 1D random generator
    return fract(sin(seed) * 43758.5453123);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // 1. Get the screen coordinates (Code 1 actually centers them at 0,0 right here!)
    vec2 uv = (fragCoord.xy - iResolution.xy*0.5)/iResolution.y;
    
    // =========================================================================
    // PART 1: APPLY ZOOM AND ROTATE TO THE UV COORDINATES
    // =========================================================================
    
    float MAX_ZOOM = 0.10;  
    float MAX_ANGLE = 0.03; 
    
    float beatTimer = floor(iTime * 4.0);
    float rDir = (rand(beatTimer) > 0.5) ? 1.0 : -1.0; 
    float rMult = 1.0 + (rand(beatTimer + 5.0) * 0.2); 
    
    // Calculate current scales based on audio sync
    float currentScale = 1.0 + (MAX_ZOOM * iMusicSync0);
    float currentAngle = MAX_ANGLE * iMusicSync1 * rDir * rMult;

    // Because Code 1's formula already centers 'uv' to (0,0), 
    // we DON'T need to subtract and add 0.5 like we did with the texture!
    
    // Apply Scale (Zoom)
    uv /= currentScale;
    
    // Apply Rotation
    float s = sin(currentAngle);
    float c = cos(currentAngle);
    mat2 rotMat = mat2(c, -s, 
                       s,  c);
    uv = rotMat * uv; // The canvas is now zoomed and rotated


    //vec2 uv = fragCoord/iResolution.xy;
    //uv = 2.0*(uv-0.5);
    //uv.y *= iResolution.y/iResolution.x;
        uv = rot(uv,TK*iTime);
	vec3 ro = vec3(0.0,0.0,0.1);
	vec3 rd = normalize(vec3(uv,0.0)-ro);
	float t  = 2.0;
	float d = 0.0;
	float ac = 0.0;
	for(int i = 0; i < 66; i++){
		d = dist(ro+rd*t)*0.2;
		d = max(0.0000,abs(d));
		t += d;
		if(d<0.001) ac += 0.1; //exp(-15.0*d);
	}
	
	// Create a dynamic rainbow color based on depth (t) and time (iTime)
	// vec3(0.0, 2.0, 4.0) offsets the RGB channels to create the rainbow spectrum
	vec3 rainbow = 0.5 + 0.5 * cos(iTime * 2.0 + t + vec3(0.0, 2.0, 4.0));
	
	vec3 col = vec3(0.0);
	// Apply the rainbow to the base color
	col = rainbow * 0.2 * vec3(ac);
	
	vec3 pn = ro+rd*t;
	float kn = 0.5;
	pn.z += -1.5*iTime*TK;
	pn.z = mod(pn.z,kn)-0.5*kn;
	float em = clamp(0.01/pn.z, 0.0, 100.0);
	
	// Apply the rainbow to the emission (glow) color
	col += 3.0 * em * rainbow;
	
	col = clamp(col,0.0,1.0);


    // Output to screen
    fragColor = vec4(col,0.);



    // =========================================================================
    // PART 2: FLASHING COLORS
    // =========================================================================
    

    // MUSIC_VIDEO__MOD
    float colorTimer = floor(iTime * 2.0); 

    vec3 flashColor = vec3(
        rand(colorTimer),           
        rand(colorTimer + 10.0),    
        rand(colorTimer + 20.0)     
    );

    // Add the flash to the transformed screen
    vec3 finalColor = fragColor.rgb + (flashColor * iMusicSync2*1.2);

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 

    // --- UNIVERSAL FLASH TO WHITE (FIXED TIMING) ---
    // 1. Define exactly how long this animation should take (e.g., 2.5 seconds)
    float transitionTime = 1.0; 

    // 2. Create a timer that stays at 0.0 for the whole video, 
    // but starts counting up exactly 'transitionTime' seconds before the end.
    float triggerTime = iEndTime - transitionTime;
    float tFlash = clamp(iTime - triggerTime, 0.0, transitionTime);

    // 3. Run your exact same math, but scale it to 'transitionTime' instead of 'duration'
    float z_prog = smoothstep(0.0, transitionTime, tFlash);
    z_prog = pow(z_prog, 1.5); // Easing for accelerating zoom effect
    
    float animZ = mix(20.0, -2.0, z_prog);
    float flashAmt = 1.0 - smoothstep(-1.0, 1.0, animZ);
    
    // 4. Mix and output
    vec3 colFlash = mix(fragColor.rgb, vec3(1.0), flashAmt);
    fragColor = vec4(colFlash, 1.0);


}
