/*
    "Chromatic Aberration" by @XorDev
    
    Here's a super simple "Chromatic Aberration" effect that you can easily
    add as a post-process effect to other shaders.
    
    Used here: https://www.shadertoy.com/view/mdfGRs
*/


#define pi 3.14159
#define oz vec2(1,0)

// random [0,1]
float hash12(vec2 p) {
	vec3 p3  = fract(vec3(p.xyx) * .1031);
	p3 += dot(p3, p3.yzx + 33.33);
	return fract((p3.x + p3.y) * p3.z);
}

// random normalized vector
vec2 randVec(vec2 p) {
	float a = hash12(p);
	a *= 2.0*pi;
	a += iTime;
	return vec2(sin(a), cos(a));
}

// perlin noise
float perlin_bioscanner(vec2 p) {
	vec2 f = fract(p);
	vec2 s = smoothstep(0.0, 1.0, f);
	vec2 i = floor(p);
	float a = dot(randVec(i), f);
	float b = dot(randVec(i+oz.xy), f-oz.xy);
	float c = dot(randVec(i+oz.yx), f-oz.yx);
	float d = dot(randVec(i+oz.xx), f-oz.xx);
	return mix(mix(a, b, s.x), mix(c, d, s.x), s.y)/0.707107;
}

// fractal noise
float fbm(vec2 p) {
	float a = 0.5;
	float r = 0.0;
	for (int i = 0; i < 8; i++) {
		r += a*perlin_bioscanner(p);
		a *= 0.5;
		p *= 2.0;
	}
	return r;
}

// 2D square sdf
float square(vec2 p, vec2 s) {
	vec2 d = abs(p)-s;
	return length(max(d, 0.)) + min(0., max(d.x,d.y));
}


void mainImage(out vec4 fragColor, vec2 fragCoord)
{
    //Compute uv coordinates from fragcoordinates
    vec2 u = fragCoord/iResolution.xy;
    //Color output starts at 0.
    vec4 c = vec4(0);
	
    //Iterate 20 times from 0 to 1
	for(float i = 0.; i<1.; i+=.05)
        //Add a texture sample approaching the center (0.5, 0.5)
        //This center could moved to change how the direction of aberation
        //The mix amount determines the intensity of the aberration smearing

	c += texture(iChannel0, mix(u, vec2(.5), i*.2*cos(iTime*3.0))).rgba*
        
        //This makes each sample have a different color from red to green to blue
        //The total should be multiplied by the 2/number of samples, (e.g. 0.1)
        vec4(i,1.-abs(i+i-1.),1.-i,1)*.1;
	
    //Output the resulting color
    fragColor = c;



	// coordinates
	vec2 uv = fragCoord.xy/iResolution.y;
	uv.y = 1.-uv.y;
	float ar = iResolution.x/iResolution.y;
	uv -= vec2(ar/2.0,0.5);
	float uxy = uv.x+uv.y;

	// distortion noise
	float n = fbm(vec2(10.*uv.y, 3.*iTime));
	float g = 0.2*n-1.9+sin(0.3*iTime);
	uv.x += smoothstep(0.15, 0., abs(g))*n;

	// seeking box target position
	float s = perlin_bioscanner(vec2(0.3*iTime));
	vec2 p = vec2(perlin_bioscanner(vec2(0.5*iTime)), 0.5*perlin_bioscanner(vec2(-iTime)))*0.5*(iResolution.xy - vec2(300, 300))/iResolution.y;
	p.y -= 0.1*sin(6.*p.x+iTime);
    
    // Empty base color
    float r = 0.;

    // --- LINE THICKNESS SETTINGS ---
    // Increase these values to make the lines wider. 
    // 0.0 is the original 1-pixel width.
    float outerBoxThickness   = 0.001; 
    float crosshairThickness  = 0.002;
    float seekingBoxThickness = 0.004; // <-- The moving box width

	// ui
	float ob = square(uv, 0.5*vec2(0.9*ar,1.-0.1*ar));
    
    // Outer border
	float ui = 1.5*smoothstep(outerBoxThickness + fwidth(uxy), outerBoxThickness - fwidth(uxy), abs(ob));
    
	float ms = step(ob, -0.005) * smoothstep(0.01, 0.005, abs(abs(uv.y)-0.475*(1.-0.1*ar)));
	ms *= smoothstep(2.*fwidth(uv.x), fwidth(uv.x), fract(uv.x*100.)/100.);
	ui = max(ui, ms);
    
    // Crosshair lines inside the seeking box
	float cr = smoothstep(crosshairThickness + fwidth(uxy), crosshairThickness - fwidth(uxy), abs(min(abs(uv.x-p.x), abs(uv.y-p.y))));
    
    // The seeking box shape
	cr = max(cr, 3.*smoothstep(seekingBoxThickness + fwidth(uxy), seekingBoxThickness - fwidth(uxy), abs(square(uv-p, (0.1+smoothstep(-0.25, 0.5, s))*vec2(0.15)))));
	
    cr *= smoothstep(0., -0.1, ob);
	ui = max(ui, cr);
    
    // combine
	r = max(r, 2.*ui);
	r *= 2.*pow(0.5+0.5*n, 2.);
	r *= 0.5+0.5*hash12(fragCoord.xy+iTime);
	r = clamp(r, 0., 1.);
    
	vec4 finalColorr = vec4(mix(r, pow(1.-r, 3.), smoothstep(-0.025, 0.025, g)));

    fragColor.rgb = mix(fragColor.rgb, finalColorr.rgb, finalColorr.a);

    // Calculate the intensity of the UI action (how bright the lines/glitches should be)
    float uiIntensity = mix(r, pow(1.-r, 3.), smoothstep(-0.025, 0.025, g));
    
    // Generate a dynamic rainbow color that shifts based on time and UV coordinates
    vec3 rainbowColor = 0.5 + 0.5 * cos(iTime * 3.0 + (uv.x + uv.y) * 4.0 + vec3(0.0, 2.0, 4.0));

    fragColor.rgb = mix(fragColor.rgb, rainbowColor, uiIntensity);


}
