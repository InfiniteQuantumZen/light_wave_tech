#define T iTime
#define P(z) vec3(2e2 * cos( (z)*vec2(.003,.0033) ) , z)
#define O(Z,c) ( length(                 /* orb */   \
          p - vec3( X.x+sin( T*c*6. ) * 6.,        \
                    X.y+sin( T*c*4. ) * 1. + 1e1,  \
                    T*3e2+Z+4e1+2e1*cos(T*.6) )  ) - c )
float noise(vec3 p) {
    float a,s;
    s = 6e1 - length(p.xy);
    a = 1.;
    s -= abs(dot(sin(.1*p.z+T+.2*p / a ), p-p+a*1.));
    s -= abs(dot(sin(.2*p.z+T+.3*p / a ), p-p+a*2.));
    s -= abs(dot(sin(.3*p.z+T+.4*p / a ), p-p+a*4.));
    return s;
}


void mainImage(out vec4 o, vec2 u) {

    float e,i,s;
    
    vec3  r = iResolution;
    u = (u + u - r.xy) / r.y;
    //if (abs(u.y) > .8) { o*= i; return;}
    
    vec3  p = P(T*3e2),
          Z = normalize( P(T*3e2+1e1) - p),
          X = normalize(vec3(Z.z,0,-Z)),
          D = vec3(mat2(cos(cos(.3*T)*.5+vec4(0,33,11,0))) * u, 1)
                   * mat3(-X, cross(X, Z), Z);
                   
    vec2 v = (.6+.6*sin(T))+u + (u.yx*.8+.2-vec2(-1.,.1));
    for(o *= i; i++<64.;
        s = .2+ .7*abs(min(e,max(noise(Z - X),s))),
        o += 1./s + 1e1*vec4(1,2,9,0)/max(e, .01) + .6*vec4(1,2,3,0)/length(v)
    )
        Z = p += D * s,
        X = P(p.z),
        e = max( min( O( 3., .1),
                 min( O( 5., .2),
                      O( 9., .3) )), .01),
        Z += cos(Z.yzx),
        s = 1e2*dot(sin(Z/1e2), cos(Z.yzx/2e2)),
        s -= dot(cos(Z*4.), Z-Z+.5);
    
    o = mix(o, o.yxzw, smoothstep(.2, 1., length(u)/2.));
    o = tanh(o*o/3e4);

}
