void mainImage( out vec4 O, vec2 I ){
    vec3 p, q, r = normalize(vec3(I+I,0) - iResolution.xyy);
    float i, j, t, v, s;
    // Raymarching loop, bound maximum step size
    for (O*=i; i++<50.;t+=min(v*.15,1.)){
    p=t*r;
    
    // Move camera back
    p.z+=2.5;
    
    // Rotate different amount each y layer
    p.xz*=mat2(cos((sin(round(p.y*3.)*2.)+.2)*iTime+vec4(0,11,33,0)));
    
    // Start with density based on solid hyperboloid
    q=p*p;
    s=q.x+q.z-q.y;
    v = max(s-1.,.01);
    
    // Add two hyperboloid surfaces
    v = min(v,abs(s-2.)+.001);
    v = min(v,abs(s-7.)+.001);
    
    // Repeatedly carve cube shells out of structure
    s=1.;
    for(j=0.;j++<6.;){
        p*=1.5;
        s*=1.5;
        q=abs(mod(p-1.,2.)-1.);
        v=max(v,(.2-abs(.6-max(max(q.x,q.y),q.z)))/s);
    }
    
    // Color accumulation based on distance travelled and density
    O+=exp(cos(t+vec4(0,.5,1.5,0)))/sqrt(v);
    }
    // Tone mapping
    O=tanh(O*O/4e4);
}
