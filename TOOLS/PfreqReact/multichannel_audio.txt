def get_wasapi_71_device_id(device_name="ROG Theta"):
    """
    Scans all audio devices and returns the ID of the WASAPI device
    that matches our headset and supports 8 output channels.
    """
    devices = sd.query_devices()
    for i, dev in enumerate(devices):
        # Find out the name of the Host API (e.g., MME, DirectSound, WASAPI)
        hostapi_name = sd.query_hostapis(dev['hostapi'])['name']
        
        # Look for our specific combination
#        if "WASAPI" in hostapi_name and device_name in dev['name'] and dev['max_output_channels'] >= 8:
#            return i

        if "Windows DirectSound" in hostapi_name and device_name in dev['name'] and dev['max_output_channels'] >= 8:
            return i
            
    return None # Fallback if not found

    meter = pyln.Meter(rate)
    current_loudness = meter.integrated_loudness(data)
    normalized_audio = pyln.normalize.loudness(data, current_loudness, target_lufs)
    normalized_audio = np.clip(normalized_audio, -1.0, 1.0)
    
    # 2. Build the 8-channel array (Same as before)
    # sounddevice expects float32 arrays natively, which is great!
    padded_audio = np.zeros((normalized_audio.shape[0], 8), dtype=np.float32)
    
    left = normalized_audio[:, 0]
    right = normalized_audio[:, 1]
