// iChannel1: "RGBAnoise_medium_256x256.png"

#define r iResolution.xy
const float PI = 3.14;

void mainImage(out vec4 c, vec2 p) {
    p = (p + p - r) / sqrt(r.x * r.y) * 0.1; 
    float f = length(p) - 0.001 / length(p);
    p -= p / dot(p,p) * 0.001;
    
    c = vec4(0,0,0,1);
    for(float i = 1.0; i < 14.0; i++) {
        vec2 k = vec2(atan(p.y,p.x) / PI * 1.0 + 100.0,0.003 / length(p)) * i;
        k.y += iTime * 0.8 * sign(f);
        for(float j = 0.0; j < 16.0; j++) {
            vec2 o = fract(k + j / 12.0) - (texture(iChannel1,floor(fract((k + j / 16.0) / 2.0) * 2.0 + i * 7.0 + j * 4.0) / 256.0 + j * 137.0 / 256.0).xy * 0.9 + 0.05);
            c.xyz += min(pow(max(0.3 / (length(o) / 0.08) - 0.01,0.0),2.0) * (sin((j * 0.25 - i * 0.65) + iTime + vec3(0,2,4) / 3.0 * PI) * 0.5 + 0.5),3.5) / (0.8 * (k.y - iTime * 0.8 * sign(f)) + 1.0) * (1.0 - i / 14.0);
        }
    }
}
