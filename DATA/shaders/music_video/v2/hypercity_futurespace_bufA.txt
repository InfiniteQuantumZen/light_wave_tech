#define O(Z,c) ( length(                 /* orb */   \
          p - vec3( sin( T*c*4. ) * 16.,        \
                    sin( T*c*6. ) * 16.,  \
                    Z+2e1+1e1*cos(T*.6) )  ) - c )
// MENGERLAYER
#define m(f, h)\
    s /= (f), \
    p = abs(fract(q/s)*s - s*.5), \
 	m = min(m, min(max(p.x, p.y), \
               min(max(p.y, p.z), \
               max(p.x, p.z))) - s/(h))

void mainImage(out vec4 o, vec2 u) {
   
    float i, e, T = iTime * 2.,m,d,s,pi = 3.14159;
    vec3  c,r = iResolution,
          a = cos(vec3(2,3,1)+iTime/4.);
    
    u = (u+u - r.xy) / r.y;

    vec3  q,p = vec3(0),
          D = vec3(u, 1);
 
    for(;i++ < 1e2;
        d += s = min(e,
                 max(.01+.6*abs(m), .01+.6*abs(2.+2.*dot(sin(.3*T+p/6.), cos(p.yzx/16.))))),
        c += (1.+cos(.1*i+vec3(2,1,0)))/s + 2e1*vec3(1,2,3)/max(e, .001),
        p = q
    )
        for(q = p += D * s,
            e = max(  min( O( 3., .1),
                      min( O( 5., .2),
                           O( 9., .3) )), .001),
            m=.1,
            s = 256.,
            m(2., 4.),
            s = 64.,
            m(12., 4.),
            s = 32.,
            m(24., 4.),
            p = q,
            p = a*dot(a,p)-cross(a,p),
            s = .075; s < 1e1; s *= 2.3
        )
            p.xz += (pi * ( abs( 2. * fract( s*p/(2.*pi)+.75)-1.)-.5)).zx/s;

    o.rgb = tanh(c*c/4e6  * exp(d/5e1));

}
