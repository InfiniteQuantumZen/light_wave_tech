// issue: 
// floor(val) + ipos gives it a diagonal bias (not always noticeable)

float h21 (vec2 a) {
    return fract(sin(dot(a.xy, vec2(12.9898, 78.233))) * 43758.5453123);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = (fragCoord-0.5*iResolution.xy)/iResolution.y;
    
    float sc = 24.;
    vec2 ipos = floor(sc * uv) + 0.5;
    vec2 fpos = sc * uv - ipos;
    
    // you can put anything here
    float val = h21(ipos) - iTime;
    // val = h21(ipos) - 10. * uv.y - iTime;
    
    // Gives each cell a fixed angle (removes diagonal bias)
    /*
    float a = 2. * 3.14159 * h21(ipos);
    float c = cos(a); 
    float s = sin(a);
    
    // current value (using 0.01 so it looks "random" for longer)
    float v = h21(vec2(c * floor(val), s * floor(val)) + 0.01 * ipos);
    // next value
    float v2 = h21(vec2(c * (floor(val) + 1.), s * (floor(val) + 1.)) + 0.01 * ipos);
    */
    
    // current value (using 0.01 so it looks "random" for longer)
    float v = h21(floor(val) + 0.01 * ipos);
    // next value
    float v2 = h21(floor(val) + 1. + 0.01 * ipos);
    
    // smooth lerp between values
    float m = fract(val);
    m = m * m * (3. - 2. * m);
    v = mix(v, v2, m);
    
    fragColor = vec4(v);
}
