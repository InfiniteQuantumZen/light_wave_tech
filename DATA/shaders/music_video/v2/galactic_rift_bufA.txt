void mainImage(out vec4 o, vec2 u) {
    float d,a,i,s,t = iTime*5.+sin(iTime/4.)*2.;
    vec3  p = iResolution,
          D = normalize(vec3(u = (u+u-p.xy)/p.y, .7));
    u += cos(t*vec2(.23,.15))/3.;
    vec2 v = (.2+.2*sin(t*.6))+u + (u.yx*.8+.2-vec2(-1.,.1));
    for(o*=0.; i++<5e1 && d < 1e2;
        s = max(s, sin(p.z)*5.+16. - abs(p.y)),
        d += s = .01+.4*abs(s),
        o += .2*vec4(12,3,6,0)/length(v)
          + (1.+cos(.2*p.z+vec4(3,1,0,0)))/(1e1*s)
          + vec4(1,2,7,0)/length(u)/6e1
    )
        for(p = D*d, p.z += t,
            s = 6.-abs(p.x),
            p.xy *= mat2(cos(.02*p.z+.04*t+vec4(0,33,11,0))),
            p.z *=.7,
            a = .01; a < .5;
            p -= dot(ceil(cos(.3*p / a)), vec3(4.)) * a,
            p += ceil(sin(p.yzx*.3)*3.),
            a += a);
    o = tanh(o*o/1e4);
}
