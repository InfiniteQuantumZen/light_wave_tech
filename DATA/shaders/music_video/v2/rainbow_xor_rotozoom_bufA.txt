#define PI 3.1415926535897932384626433832795

mat2 rotate2d(float angle) {
	return mat2(cos(angle), -sin(angle),
				sin(angle),  cos(angle));
}

vec2 Rotate(vec2 v, float angle) {
    return v*rotate2d(angle);
}

// https://www.shadertoy.com/view/wt23Rt

//Hue to RGB (red, green, blue).
//Source: https://github.com/tobspr/GLSL-Color-Spaces/blob/master/ColorSpaces.inc.glsl
#ifndef saturate
#define saturate(v) clamp(v,0.,1.)
//      clamp(v,0.,1.)
#endif
vec3 hue2rgb(float hue){
	hue=fract(hue);
	return saturate(vec3(
		abs(hue*6.-3.)-1.,
		2.-abs(hue*6.-2.),
		2.-abs(hue*6.-4.)
	));
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // Normalized pixel coordinates (from 0 to 1)
    vec2 uv = (2.0*fragCoord-iResolution.xy)/iResolution.y;

    uv = Rotate(uv,iTime*PI/12.+length(uv)*sin(iTime/3.));
    int col = (int(uv.x*256.)^int(uv.y*256.))&255;

    fragColor = vec4(hue2rgb((float(col)/256.+iTime/5.)+2./length(uv)+iTime/3.),1.0);
}
