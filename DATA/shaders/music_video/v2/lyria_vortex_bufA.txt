// zucconis spectral palette https://www.alanzucconi.com/2017/07/15/improving-the-rainbow-2/
vec3 bump3y (vec3 x, vec3 yoffset)
{
    vec3 y = 1. - x * x;
    y = clamp((y-yoffset), vec3(0), vec3(1));
    return y;
}
// Zucconi's spectral palette
vec3 spectral_zucconi6(float x) {
    x = fract(x);
    const vec3 c1 = vec3(3.54585104, 2.93225262, 2.41593945);
    const vec3 x1 = vec3(0.69549072, 0.49228336, 0.27699880);
    const vec3 y1 = vec3(0.02312639, 0.15225084, 0.52607955);
    const vec3 c2 = vec3(3.90307140, 3.21182957, 3.96587128);
    const vec3 x2 = vec3(0.11748627, 0.86755042, 0.66077860);
    const vec3 y2 = vec3(0.84897130, 0.88445281, 0.73949448);
    return bump3y(c1 * (x - x1), y1) + bump3y(c2 * (x - x2), y2) ;
}

vec3 getColorRamp(int palette, float x) {

    x = fract(x);
    vec3 colors[10];
    int len;
    
    
    if(palette == 0) {
        len = colors1.length();
        COPY_COLOR(len, colors1);       
    }
    
    if(palette == 1) {
        len = colors2.length();
        COPY_COLOR(len, colors2);
    }
    if(palette == 2) {
        len = colors3.length();
        COPY_COLOR(len, colors3);
        
    }
    if(palette == 3) {
        len = colors4.length();
        COPY_COLOR(len, colors4);    
    }
    
    if(palette == 4) {
        len = colors5.length();
        COPY_COLOR(len, colors5);    
    }
    if(palette == 5) {
        len = colors6.length();
        COPY_COLOR(len, colors6);    
    }
    if(palette == 6) {
        len = colors7.length();
        COPY_COLOR(len, colors7);    
    }
    
    if(palette == 7) {
        len = colors12.length();
        COPY_COLOR(len, colors12);    
    }
    if(palette == 8) {
        len = colors13.length();
        COPY_COLOR(len, colors13);    //0 + 7 
         
    }
    
    
    int colorCount = len;
    float stepSize = 1.0/float(colorCount);
    // from MadMapper gradient
    int stepId = int(x/stepSize);
    vec3 col = mix(colors[stepId%colorCount],colors[(stepId+1)%colorCount],pow(x/stepSize-float(stepId),pow(1.0,6.)));
    return col;
   // return getColorRamp_(colors, len, x);

}


vec3 getPal(int pal, float x) {

    return getColorRamp(pal, x);
    // Wrap x to the 0-1 range
    float wrappedX = fract(x);

    // Determine the blend factor for the wrap-around edge
    float blendFactor = smoothstep(0.7, 1.3, x);

    // Get colors from the ramp
    vec3 currentColor = getColorRamp(pal, wrappedX);
    vec3 nextColor = getColorRamp(pal, fract(wrappedX + 0.3));

    // Blend the colors smoothly
    return mix(currentColor, nextColor, blendFactor);
    
    return currentColor;
}


mat2 rot(float x) {return mat2(cos(x), -sin(x), sin(x), cos(x));}
vec3 pal(float x) {return .5+.5*cos(6.28*x-vec3(5,0,2));}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = (fragCoord - .5*iResolution.xy)/iResolution.y;
    float tt = iTime*.5;
    
    uv.xy *= mix(.8, 1.2, SIN(-tt+5.*length(uv.xy)));
    
    vec3 col = vec3(0);
    vec3 rd = vec3(uv, 1);
    vec3 p = vec3(0);
    float t =.1*texelFetch(iChannel0, ivec2(fragCoord)%1024, 0).a; 

    
    for(float i=0.; i<120.; i++) {
        vec3 p = t*rd + rd;
        p.z += tt;
       // p = abs(p)-.3;
        float z = p.z;
        p.xy *= rot(p.z);
      ///  p = abs(p)-.3;
    
       for(float j=0.; j < 3.; j++) {     
            float  a= exp(j)/exp2(j);
            p += cos(3.*p.yzx*a + .5*tt - length(p.xy)*9.)/a; 
        }
        float d = 0.007 + abs((exp2(1.3*p)-vec3(0,1.+.7*SIN(tt),0)).y-1.)/14.;
  //     float d = 0.007 + abs(((1.3*p)-vec3(0,1.+.7*SIN(tt),0)).y-1.)/14.;
        float k = t*.7 +length(p)*.1 - .2*tt + z*.1;
        vec3 c = getPal(7, k);
        c = mix(c, c*vec3(0.922,0.973,0.725), SIN(z*.5));
        col += c*1e-3/d;       
        t += d/4.;
    }
   
    float gl = exp(-20.*length(uv.xy));
    col += .4*mix(vec3(0.361,0.957,1.000), vec3(0.847,1.000,0.561), SIN(gl*2.-tt))*pow(gl*11., 1.);
    
    col *= tanh(col*.1);
    col = pow(col, vec3(.45));
    fragColor = vec4(col,1.0);
}
