#define gridify(uv) smoothstep(-0.002,0.0,uv)

// slightly altered https://www.shadertoy.com/view/wtBXRz
vec2 brownConradyDistortion(in vec2 uv, in float k1)
{
    uv = uv * 2.0 - 1.0;	// brown conrady takes [-1:1]

    // positive values of K1 give barrel distortion, negative give pincushion
    float r2 = uv.x*uv.x + uv.y*uv.y;
    uv *= 1.0 + k1 * r2;
    
    uv = (uv * .5 + .5);	// restore -> [0:1]
    return uv;
}

float grid(in vec2 uv){
    vec2 fw = fwidth(uv);
	float xx = smoothstep(-fw.x,fw.x,uv.x) * smoothstep(-fw.x,fw.x,1.0-uv.x);
	float xy = smoothstep(-fw.y,fw.y,uv.y) * smoothstep(-fw.y,fw.y,1.0-uv.y);
	return xx*xy;
}

// 1. Put this function ABOVE your mainImage function
float rand(float seed) {
    // This is the classic GLSL 1D random generator
    return fract(sin(seed) * 43758.5453123);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // Normalized pixel coordinates (from 0 to 1)
    vec2 uv = fragCoord/iResolution.xy;
    float DISTORT = sin(iTime)*0.05;
    
    uv = brownConradyDistortion(uv, DISTORT*0.9);
    uv.y = 1.0 - uv.y;
    vec2 texuv = uv+vec2(iTime*0.001, 0.0);
    
    
    vec2 yo;
    yo.x = (uv.x-0.5)*-DISTORT;
    yo.y = (uv.x-0.5)*DISTORT;
	
    vec2 xo;
    xo.x = (uv.y-0.5)*-DISTORT;
    xo.y = (uv.y-0.5)*DISTORT;
    
    vec3 col;
    col.r = texture(iChannel0,texuv+vec2(yo.x,xo.x)).r;
    col.g = texture(iChannel0,texuv).r;
    col.b = texture(iChannel0,texuv+vec2(yo.y,xo.y)).r;
	
    col *= grid(uv);
    
    fragColor = vec4(col,1.0);



    // MUSIC_VIDEO__MOD
    float colorTimer = floor(iTime * 2.0); 

    vec3 flashColor = vec3(
        rand(colorTimer),           
        rand(colorTimer + 10.0),    
        rand(colorTimer + 20.0)     
    );

    // Add the flash to the transformed screen
    vec3 finalColor = fragColor.rgb + (flashColor * iMusicSync2*1.2);

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 
}
