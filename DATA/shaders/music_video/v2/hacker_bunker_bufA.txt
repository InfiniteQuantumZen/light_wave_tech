#define T (iTime)

#define W 3.
#define H 1.5

#define R(a) mat2(cos(a + vec4(0,33,11,0)))
#define N normalize
#define inf 9e9

#define O(Z,c) ( length( \
          p - vec3( sin( T*c*2. )/2.,        \
                    sin( T*c*4. )/2.,  \
                    T+T+ 1e1)  ) - c/12. )

#define MENGERLAYER(scale, minmax, hole)\
    s /= (scale), \
    p = abs(fract(q/s)*s - s*.5), \
 	d = minmax(d, min(max(p.x, p.y), \
                  min(max(p.y, p.z), \
                  max(p.x, p.z))) - s/(hole))
float fractal(in vec3 q){
    vec3 p;
    float d=inf, s = 2.;
    MENGERLAYER(1., min, 4.);
    MENGERLAYER(3., max, 3.);
    return d;
}

float tunnel(vec3 p) {
    vec3 tun = abs(p);
    return max(W - tun.y - tun.x,
           min(W -tun.x, H-tun.y));
}

vec4 lights;
float map(vec3 p) {
    
    float l;

    l = min( O( 1., .4),
        min( O( 2., .5),
             O( 3., .6) )),
    l = min(.001+abs(l), .02+abs(length(p.xy -  sin(p.z/1.5  + vec2(0, 1.57))) - .05));

    lights += vec4(1,2,1.1,0)/l;

    return min(l, max(tunnel(p), fractal(p)));
}


void mainImage(out vec4 o, in vec2 u) {
    float i,s,d;
    vec3  r = iResolution;
    
    u = (u-r.xy/2.)/r.y;
    
    o = vec4(0);
    vec3  p,D = N(vec3(R(sin(T*.005)*.4)*u, 1));
    
    for(; i++ < 70.;)
        p = D * d,
        p.z += T+T,
        d += s = map(p)*.8,
        o += pow(lights, vec4(2)) + 1./max(s, .01);


    // normal
    // tetrahedron technique: https://iquilezles.org/articles/normalsSDF/
    const float h = 0.001;
    const vec2 k = vec2(1,-1);
    vec3 n = N(k.xyy*map( p + k.xyy*h ) + 
               k.yyx*map( p + k.yyx*h ) + 
               k.yxy*map( p + k.yxy*h ) + 
               k.xxx*map( p + k.xxx*h ) );

    // diffuse
    o *= (.1 + max(dot(n, -D), 0.));
    
    // reflection march
    vec4 ref;
    lights = vec4(0);
    for(p += n*.05, D = reflect(D, n), s=i=0.; i++<4e1; )
        p += D*s,
        s = map(p)*.8,
        ref +=  pow(lights, vec4(1.2)) + 1./max(s, .01);

    o += o*ref;
    o = tanh(o / 6e8 / exp(d/1e1));
}
