// --- SDF Helper Functions ---

// Calculates distance to a line segment (x) and the 1D distance ALONG the segment (y)
vec2 sdLineDist(vec2 p, vec2 a, vec2 b) {
    vec2 pa = p - a, ba = b - a;
    float l = length(ba);
    if (l < 0.0001) return vec2(length(pa), 0.0);
    float h = clamp(dot(pa, ba) / (l * l), 0.0, 1.0);
    return vec2(length(pa - ba * h), h * l);
}

// Combines 3 connected segments into a single track
// Returns vec3: (distance to track, distance along track, total length of track)
vec3 circuitSDF(vec2 p, vec2 p1, vec2 p2, vec2 p3, vec2 p4) {
    vec2 d1 = sdLineDist(p, p1, p2);
    vec2 d2 = sdLineDist(p, p2, p3);
    vec2 d3 = sdLineDist(p, p3, p4);

    float l1 = length(p2 - p1);
    float l2 = length(p3 - p2);
    float l3 = length(p4 - p3);

    // Offset distances for 2nd and 3rd segments to make the 1D length continuous
    d2.y += l1;
    d3.y += l1 + l2;

    // Find the closest segment to our current pixel
    vec2 minD = d1;
    if(d2.x < minD.x) minD = d2;
    if(d3.x < minD.x) minD = d3;

    return vec3(minD.x, minD.y, l1 + l2 + l3);
}

// --- Circuit Rendering Logic ---

vec3 renderCircuit(vec2 uv, vec2 p1, vec2 p2, vec2 p3, vec2 p4, float delay, vec3 color, float localTime) {
    vec3 t = circuitSDF(uv, p1, p2, p3, p4);
    float d = t.x;      // 2D distance to the line
    float l = t.y;      // 1D distance along the path
    float totalL = t.z; // Total length of the trace

    // Define "Nodes" (the little circles at the ends of tracks)
    float rNode = 0.002;
    float dNode1 = length(uv - p1) - rNode;
    float dNode4 = length(uv - p4) - rNode;

    // 1. Draw Backbone (Outline / Darker Track) Animation
    float drawProgress = max(0.0, localTime * 1.5 - delay);
    float isDrawnStep = step(l, drawProgress); // 1.0 if track is drawn, 0.0 if not
    
    float node1Drawn = step(0.0, drawProgress);
    float node4Drawn = step(totalL, drawProgress);
    
    // Track styling
    float backboneWidth = 0.0008;
    float backboneAlpha = smoothstep(backboneWidth + 0.003, backboneWidth, d) * 0.25;
    backboneAlpha += smoothstep(0.015, 0.0, d) * 0.08; // subtle outer glow
    backboneAlpha *= isDrawnStep;
    
    // Node hollow rings
    float node1Alpha = smoothstep(0.003, 0.001, abs(dNode1)) * 0.5 * node1Drawn;
    float node4Alpha = smoothstep(0.003, 0.001, abs(dNode4)) * 0.5 * node4Drawn;
    backboneAlpha += node1Alpha + node4Alpha;
    
    // 2. Traveling Pulse / Current Animation
    float pulseSpeed = 1.2;
    float pulseCycle = totalL + 1.5; // add some pause between pulses
    // Start pulses flowing based on global iTime so they never stop organically
    float pulsePos = mod(iTime * pulseSpeed - delay, pulseCycle) - 0.2; 
    
    float pulseDist = l - pulsePos; // Distance from the pulse head
    float pulseLength = 0.3;        // How long the light tail is
    
    // Shape the pulse: sharp head, fading tail
    float pulseIntensity = 0.0;
    if (pulseDist < 0.0 && pulseDist > -pulseLength) {
        pulseIntensity = 1.0 - (abs(pulseDist) / pulseLength);
        pulseIntensity = pow(pulseIntensity, 3.0); 
    }
    
    // Hide pulse if it's on a section of track that hasn't been drawn yet
    pulseIntensity *= isDrawnStep;
    
    // Add glow for the pulse
    float pulseWidth = 0.0015;
    float pulseGlow = smoothstep(pulseWidth + 0.006, pulseWidth, d) * pulseIntensity * 1.5;
    pulseGlow += smoothstep(0.02, 0.0, d) * pulseIntensity * 0.4; // wider glow
    
    // Bright dot at the head of the pulse
    float headDist = length(vec2(d, pulseDist));
    float headDot = smoothstep(0.02, 0.002, headDist) * isDrawnStep;
    if (pulsePos > totalL || pulsePos < 0.0) headDot = 0.0; // Hide dot when off track
    pulseGlow += headDot * 1.5;
    
    // Flash the nodes when the pulse hits them
    float node1Pulse = smoothstep(0.015, 0.0, dNode1) * max(0.0, 1.0 - abs(pulsePos)*6.0) * node1Drawn;
    float node4Pulse = smoothstep(0.015, 0.0, dNode4) * max(0.0, 1.0 - abs(pulsePos - totalL)*6.0) * node4Drawn;
    
    return color * (backboneAlpha + pulseGlow + node1Pulse + node4Pulse);
}

// --- REUSABLE WAVE FUNCTION ---
// fragCoord : Current pixel coordinates
// amplitude : Height of the wave
// yCenter   : The Y axis offset in pixels
// baseColor : The RGB color of the wave
// freq      : How tight the sine wave is (0.02 from your original code)
// phaseShift: Used with time to make the waves scroll horizontally

vec3 drawAudioWave(vec2 fragCoord, float amplitude, float yCenter, vec3 baseColor, float freq, float phaseShift) {
        
    // Graceful fallback from your Pygame code: 
    // If amplitude is tiny, transition smoothly to a white line instead of snapping harshly.
    vec3 waveColor = mix(vec3(1.0, 1.0, 1.0), baseColor, smoothstep(0.0, 2.0, amplitude));

    // Calculate the mathematical Y position of the sine wave for this X pixel
    float waveY = yCenter + amplitude * sin(fragCoord.x * freq + phaseShift);
    
    // Vertical distance from the current pixel to the wave
    float dist = abs(fragCoord.y - waveY);
    
    // --- COOL FACTOR: THE GLOW ---
    // Core thickness creates a bright center, exponential falloff creates a soft neon glow
    float coreThickness = 0.0;

    if (amplitude < 2.0) {
        coreThickness = 0.1;
    }
    else {
        coreThickness = 0.5;
    }

    float glowSpread = 3.0;
    float intensity = (coreThickness / (dist + 1e-4)) + exp(-dist / glowSpread);
    
    // --- COOL FACTOR: SMOOTH EDGES ---
    // Instead of hard cuts at 450 and (Width - 400), we fade the line in and out beautifully
    float minX = 450.0;
    float maxX = iResolution.x - 400.0; 
    float mask = smoothstep(minX - 50.0, minX, fragCoord.x) * 
                 (1.0 - smoothstep(maxX, maxX + 50.0, fragCoord.x));
                 
    return waveColor * intensity * mask;
}


vec3 B2_spline(vec3 x) { // returns 3 B-spline functions of degree 2
    vec3 t = 3.0 * x;
    vec3 b0 = step(0.0, t)     * step(0.0, 1.0-t);
	vec3 b1 = step(0.0, t-1.0) * step(0.0, 2.0-t);
	vec3 b2 = step(0.0, t-2.0) * step(0.0, 3.0-t);
	return 0.5 * (
    	b0 * pow(t, vec3(2.0)) +
    	b1 * (-2.0*pow(t, vec3(2.0)) + 6.0*t - 3.0) + 
    	b2 * pow(3.0-t,vec3(2.0))
    );
}

float bump(float x) {
	return abs(x) > 1.0 ? 0.0 : 1.0 - x * x;
}



// 1. Put this function ABOVE your mainImage function
float rand(float seed) {
    // This is the classic GLSL 1D random generator
    return fract(sin(seed) * 43758.5453123);
}


void mainImage( out vec4 fragColor, in vec2 fragCoord )
{

    vec3 MAGENTA = vec3(250.0, 34.0, 197.0) / 255.0;
    vec3 GREEN   = vec3(8.0, 240.0, 124.0) / 255.0;

    vec3 GOLD    = vec3(240.0, 224.0, 8.0) / 255.0;
    vec3 PURPLE  = vec3(187.0, 34.0, 250.0) / 255.0;
    vec3 WHITE   = vec3(1.0, 1.0, 1.0);

    // =========================================================================
    // CONFIGURABLE BEZEL
    // =========================================================================

    // Clamp your audio sync value so it never exceeds 1.0 (or goes below 0.0)
    // NOT USED AT THE MOMENT float bezelIntensity1 = clamp(iMusicSync0, 0.0, 1.0);
    // NOT USED AT THE MOMENT float bezelIntensity2 = clamp(iMusicSync1, 0.0, 1.0);
    // NOT USED AT THE MOMENT float bezelIntensity3 = clamp(iMusicSync2, 0.0, 1.0);
    // NOT USED AT THE MOMENT vec3 bezelColor1 = MAGENTA * bezelIntensity1;
    // NOT USED AT THE MOMENT vec3 bezelColor2 = CYAN * bezelIntensity2;
    // NOT USED AT THE MOMENT vec3 bezelColor3 = GREEN * bezelIntensity3;
    

    float OUTER_BEZEL_WIDTH = 2.0;     // Outer rim
    float MID_BEZEL_WIDTH   = 2.0;     // Middle rim    
    float INNER_BEZEL_WIDTH = 6.0;     // Inner rim

    // Calculate cumulative distances from the edge of the screen
    float BOUND_1 = OUTER_BEZEL_WIDTH;
    float BOUND_2 = BOUND_1 + MID_BEZEL_WIDTH;
    float BOUND_3 = BOUND_2 + INNER_BEZEL_WIDTH; // This is the total thickness of all 3 combined

    // OUTER Bezel COLOR
    float colorTimerBezel = floor(iTime * 2.0);
    vec3 flashColorBezel = vec3(
        rand(colorTimerBezel + 0.0),
        rand(colorTimerBezel + 10.0),
        rand(colorTimerBezel + 20.0)
    );

    // 1. Draw the OUTER Bezel (0 to 20px)
    if (fragCoord.x < BOUND_1 || fragCoord.x > iResolution.x - BOUND_1 ||
        fragCoord.y < BOUND_1 || fragCoord.y > iResolution.y - BOUND_1) {
        fragColor = vec4(flashColorBezel * iMusicSync2*1.2, 1.0);
        return;
    }

    // 2. Draw the MIDDLE Bezel (20px to 30px)
    if (fragCoord.x < BOUND_2 || fragCoord.x > iResolution.x - BOUND_2 ||
        fragCoord.y < BOUND_2 || fragCoord.y > iResolution.y - BOUND_2) {
        fragColor = vec4(flashColorBezel * iMusicSync0*1.2, 1.0);
        return;
    }
 
    // 3. Draw the INNER Bezel (30px to 40px)
    if (fragCoord.x < BOUND_3 || fragCoord.x > iResolution.x - BOUND_3 ||
        fragCoord.y < BOUND_3 || fragCoord.y > iResolution.y - BOUND_3) {
        fragColor = vec4(flashColorBezel * iMusicSync1*1.2, 1.0);
        return;
    }

    // 4. Map coordinates for the INNER RENDER SCREEN
    // We remap the inner area back to the full 0 -> iResolution range, 
    // shrinking it proportionally past ALL THREE bezel frames.
    vec2 innerRes = iResolution.xy - 2.0 * BOUND_3;
    vec2 vCoord = (fragCoord - vec2(BOUND_3)) * (iResolution.xy / innerRes);


    // =========================================================================
    // PART 1: CIRCUIT
    // =========================================================================

    // Adjust UVs so 0,0 is the center, and y goes from -1 to 1 (maintains aspect ratio)
    // NOTE: We now use vCoord instead of fragCoord
    vec2 uv = (vCoord * 2.0 - iResolution.xy) / iResolution.y;

    vec3 finalCircuitColor = vec3(0.0);
    vec3 baseColor = vec3(1, 0.95, 1.0); // Bright white/cyan aesthetic
    
    // Create a 6-second repeating loop for the track drawing animation
    float localTime = mod(iTime, 6.0);
    
    // =========================================================================
    // --- USER ADJUSTABLE CIRCUIT PLACEMENT ---
    float circuitOffsetX = 2.50;  // Pushes circuits horizontally outwards
    float circuitOffsetY = 0.55;  // Pushes circuits vertically (if you ever need it)

    // Loop to draw tracks in all 4 quadrants of the screen
    for(int i = 0; i < 4; i++) {
        vec2 qUV = uv*1.75;
        
        // Mirror quadrants (0: TopRight, 1: TopLeft, 2: BottomRight, 3: BottomLeft)
        if(i == 1 || i == 3) qUV.x = -qUV.x;
        if(i == 2 || i == 3) qUV.y = -qUV.y;
        
        qUV.x -= circuitOffsetX;
        qUV.y -= circuitOffsetY;
        
        float qDelay = float(i) * 1.5;
        
        if (i == 1 || i == 2) {
             finalCircuitColor += renderCircuit(qUV, vec2(2.0, 0.4), vec2(1.0, 0.4), vec2(0.8, 0.2), vec2(0.3, 0.2), qDelay + 0.0, baseColor, localTime);
             finalCircuitColor += renderCircuit(qUV, vec2(0.2, 0.6), vec2(0.7, 0.6), vec2(0.9, 0.8), vec2(1.6, 0.8), qDelay + 0.5, baseColor, localTime);
             finalCircuitColor += renderCircuit(qUV, vec2(1.0, 1.1), vec2(1.0, 0.7), vec2(1.2, 0.5), vec2(1.2, 0.1), qDelay + 1.0, baseColor, localTime);
        } else {
             finalCircuitColor += renderCircuit(qUV, vec2(0.3, 0.2), vec2(0.8, 0.2), vec2(1.0, 0.4), vec2(2.0, 0.4), qDelay + 0.0, baseColor, localTime);
             finalCircuitColor += renderCircuit(qUV, vec2(1.6, 0.8), vec2(0.9, 0.8), vec2(0.7, 0.6), vec2(0.2, 0.6), qDelay + 0.5, baseColor, localTime);
             finalCircuitColor += renderCircuit(qUV, vec2(1.2, 0.1), vec2(1.2, 0.5), vec2(1.0, 0.7), vec2(1.0, 1.1), qDelay + 1.0, baseColor, localTime);
        }
    }
	
    // =========================================================================
    // PART 2: EXACT ROTOZOOM EMULATION WITH DYNAMIC PIVOTS
    // =========================================================================

    // Divide the screen coordinates by the TEXTURE'S resolution
    // NOTE: We now use vCoord instead of fragCoord
    uv = vCoord / iResolution.xy;
    
    // Convert UV to exact screen pixels (This prevents Ultrawide distortion)
    vec2 px = uv * iResolution.xy;

    // Apply the translation shake offset
    vec2 local_px = px - u_pixel_offset;

    // Prepare rotation math
    float rad = radians(u_angle_deg);
    float c = cos(rad);
    float s = sin(rad);

    // Calculate Pygame's expanded Bounding Box! 
    float bbox_w = u_target_size.x * abs(c) + u_target_size.y * abs(s);
    float bbox_h = u_target_size.x * abs(s) + u_target_size.y * abs(c);

    // --> THE SECRET SAUCE MULTIPLIER <--
    vec2 bbox_pivot = vec2(bbox_w, bbox_h) * u_pivot;

    // Shift relative to the chosen pivot
    vec2 centered_px = local_px - bbox_pivot;

    // Apply the exact Inverse Rotation Matrix that matches your orientation
    mat2 rotMat = mat2(c,-s, 
                       s, c);
    vec2 rotated_px = rotMat * centered_px;

    // Shift back from the pivot to the original unrotated surface coordinates
    vec2 surf_pivot = u_target_size * u_pivot;
    vec2 final_px = rotated_px + surf_pivot;

    // Normalize back to 0.0 - 1.0 UV space using the ZOOMED pixel dimensions
    vec2 final_uv = final_px / u_target_size;

    uv = final_uv;
    // =========================================================================
    
    vec4 texColor1 = texture(iChannel0, final_uv);
    vec2 uv_console_mask = final_uv;
    uv_console_mask.y = 1.0 - uv_console_mask.y;    
    vec4 texColor2 = texture(iChannel21, uv_console_mask);  
    fragColor.rgb = mix(texColor1.rgb, texColor2.rgb, texColor2.a);

    // NOTE: We now use vCoord instead of fragCoord
    vec2 uv_console = vCoord / iResolution.xy;
    uv_console.y = 1.0 - uv_console.y;
    vec4 texColor3 = texture(iChannel20, uv_console);
    fragColor.rgb = max(fragColor.rgb, texColor3.rgb);

    fragColor = vec4(fragColor.rgb + finalCircuitColor, 1.0);

    // we need to "cut off" the rendering if the pixels are outside our 100px box.
    // (This operates completely unaffected inside the inner bezel constraints)
    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
        fragColor = vec4(0.0, 0.0, 0.0, 0.0); // Output transparent black
        return;
    }

    // MUSIC_VIDEO__MOD
    float colorTimer = floor(iTime * 2.0); 

    vec3 flashColor = vec3(
        rand(colorTimer),           
        rand(colorTimer + 10.0),    
        rand(colorTimer + 20.0)     
    );

    // Add the flash to the transformed screen
    vec3 finalColor = fragColor.rgb + (flashColor * iMusicSync2*1.2);

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 

    // --- UNIVERSAL FLASH TO WHITE (FIXED TIMING) ---
    float transitionTime = 1.0; 

    float triggerTime = iEndTime - transitionTime;
    float tFlash = clamp(iTime - triggerTime, 0.0, transitionTime);

    float z_prog = smoothstep(0.0, transitionTime, tFlash);
    z_prog = pow(z_prog, 1.5); // Easing for accelerating zoom effect
    
    float animZ = mix(20.0, -2.0, z_prog);
    float flashAmt = 1.0 - smoothstep(-1.0, 1.0, animZ);
    
    vec3 colFlash = mix(fragColor.rgb, vec3(1.0), flashAmt);
    fragColor = vec4(colFlash, 1.0);


    // =========================================================================
    // PART 4: WAVE FUNCTION
    // =========================================================================

    // Define your colors (Normalized from 0-255 to 0.0-1.0)
    //vec3 MAGENTA = vec3(250.0, 34.0, 197.0) / 255.0;
    //vec3 GREEN   = vec3(8.0, 240.0, 124.0) / 255.0;
    //vec3 CYAN    = vec3(8.0, 224.0, 240.0) / 255.0;
    //vec3 GOLD    = vec3(240.0, 224.0, 8.0) / 255.0;
    //vec3 PURPLE  = vec3(187.0, 34.0, 250.0) / 255.0;
    //vec3 WHITE   = vec3(1.0, 1.0, 1.0);
    
    float yTop    = 1350.0;   
    float yBottom = 25.0; 

    finalColor = vec3(0.0);
    float freq = 0.02; // From your python code
    
    // --- COOL FACTOR: TIME SCROLLING ---
    float speed = iTime * 5.0;

    // NOTE: We now use vCoord instead of fragCoord
    //finalColor += drawAudioWave(vCoord, u_amp_top.x, yTop, MAGENTA, freq, speed * 1.5);
    //finalColor += drawAudioWave(vCoord, u_amp_top.y, yTop, GREEN,   freq, speed * 1.2);
    //finalColor += drawAudioWave(vCoord, u_amp_top.z, yTop, CYAN,    freq, speed * 1.0);

    finalColor += drawAudioWave(vCoord, u_amp_bottom.x*2.0, yBottom, MAGENTA, freq, speed * -3.0);

    finalColor += drawAudioWave(vCoord, u_amp_bottom.z*2.0, yBottom, CYAN, freq, speed * -3.5);

    finalColor = finalColor / (1.0 + finalColor * 0.2); 

    // Add the wave to the screen
    finalColor = fragColor.rgb + finalColor;

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 


    // =========================================================================
    // PART 6: AUDIO SPECTRUM 2
    // =========================================================================
