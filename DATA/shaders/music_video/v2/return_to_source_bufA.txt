void mainImage(out vec4 O, vec2 I) {
    vec3 p;
    float iT = iTime;
    vec3 r = normalize(vec3(I + I, 0.0) - iResolution.xyy);
    vec3 a = normalize(tan(iT * 0.06 + vec3(0.5, -0.5, 2.5)));
    
    float i = 0.0;
    float t = 0.0;
    float v = 0.0;
    float l = 0.0;
    float n = 0.0;
    
    O = vec4(-8.0);
    
    for (i = 1.0; i <= 60.0; i += 1.0) {
        p = t * r;
        
        p.z += 3.0;
        p = reflect(p, a);
        l = dot(p, p);
        
        float expansion = 0.05;
        p = log(abs(p) + expansion);
        
        vec3 flow = vec3(iT * 0.19, -iT * 0.18, iT * 0.12);
        
        for (n = 1.0; n < 9.0; n += 1.5) {
            p += 1.7 * sin((p.yzx + flow) * n) / n;
        }
        
        v = abs(length(p) - 0.9) + 0.04;
        
        O += (0.5 + 0.5 * cos(i * 0.1 + vec4(0.0, 2.0, 4.0, 0.0))) / v
           + vec4(0.8, 2.9, 1.0, 0.0) / l * 1.2;
        
        t += v * 0.053;
    }
    
    O = tanh(O / 150.0);
}
