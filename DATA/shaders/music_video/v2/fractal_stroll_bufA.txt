// just more tinkering with this fractal

#define T (iTime*1.5 + 5. + 1.5*sin(iTime*.5))
#define P(z) (vec3(cos((z) * .07) * 16., \
                   0, (z)))
#define R(a) mat2(cos(a+vec4(0,33,11,0)))
#define N normalize

float apollonian(vec3 p) {
    // FIX 1: Initialize i to 0. (s can remain uninitialized as it is written to before being read)
    float i = 0., s, w = 1.;
    vec3 b = vec3(.5, 1., 1.5);
    
    p.y -= 4.;
    p.yz = p.zy;
    p /= 18.;
    p += cos(.3*p.z+p.yzx*6.)/16.;
    
    for(; i++ < 6.;) {
        p = mod(p + b, 2. * b) - b;
        s = cos(p.z*.3)*.8+1.5 / dot(p, p);
        p *= s;
        w *= s;
    }
    return length(p) / w * 16.;
}

float map(vec3 p) {
    return max(4. - length((p - P(p.z)).xy), apollonian(p));
}

void mainImage(out vec4 o, in vec2 u) {
    // FIX 2: Initialize d (ray distance) and i (loop counter) to 0. 
    // (Removed 'a' since it was unused).
    float s = 0., d = 0., i = 0.;
    
    vec3  r = iResolution;
    u = (u-r.xy/2.)/r.y;
    
    vec3 p = P(T*2.), ro=p,
         Z = N( P(T*2.+7.) - p),
         X = N(vec3(Z.z,0,-Z)),
         D = N(vec3(R(sin(iTime*.15)*.3)*u, 1) * mat3(-X, cross(X, Z), Z));
    
    // FIX 3: Replaced the comma operator sequence with standard brackets.
    // This removes any ambiguity for strict desktop GLSL compilers.
    o = vec4(0);
    for(; i++ < 1e2;) {
        p = ro + D * d;
        s = .0008 + abs(map(p));
        d += s;
        o += (1. + cos(.07 * i + vec4(3, 1, 0, 0))) / s;
    }
    
    o = tanh(o/max(d,2e1)/2e3*exp(vec4(3,2,1,0)*d/9e1));
}
