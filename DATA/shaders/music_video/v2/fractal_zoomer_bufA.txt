// based on is shader : https://www.shadertoy.com/view/MltXz2

// z * z
vec2 zmul(vec2 a, vec2 b)
{
	//return vec2(a.x*b.x-a.y*b.y, a.x*b.y+b.x*a.y);
    return mat2(a,-a.y,a.x)*b;
}

// 1 / z
vec2 zinv(vec2 a)
{
	return vec2(a.x, -a.y) / dot(a,a);
}

const float AA = 2.;
float localTime = mod(iTime, 6.0);
    
void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    fragColor = vec4(0);
    
	vec2 g = fragCoord.xy;
	vec2 si = iResolution.xy;
    
    
    for( float m=0.; m<AA; m++ )
    for( float n=0.; n<AA; n++ )
    {
        vec2 offset = vec2(m,n) / AA - 0.5;
        vec2 uv = ((g+offset)*2.-si)/min(si.x,si.y) ;
        uv *= 30. * (sin(10.6+localTime * 0.08)*.5+.5);//zoom
        uv += vec2(-0.46022,0.746155);
        vec2 z = uv;
        vec2 c = vec2(0.66,1.23);
        float it = 0.;
        for (int i=0;i<600;i++)
        {
            z = zinv(zmul(z, z) + c);
            if( log2(log2(dot(z,z)))>1.3 ) break;
            it++;
        }
        if (it <  99.5)
        {
        	float sit = -it/pow(length(z),0.88);
            fragColor += 0.5 + 0.5*cos( 3.0 + sit*0.2 + vec4(1,0,1,1));
        }
    }
    
    fragColor /= AA * AA;
}
