void mainImage(out vec4 O, vec2 C) {
    float i,d,z;
    vec3 p,r = iResolution;
    for(O*=i;i++<1e2;O+=1./d){
        p=z*normalize(vec3(C.xy-.5*r.xy,r.y));
        p=abs(fract(vec3(mat2(cos(cos(z*.5)+vec4(0,11,33,0)))*p.xy*2.,p.z-iTime))-.5)-.5;
        z+=(d=abs(length(max(p,.25))+max(p.x,max(p.y,p.z*3.)))/3.+abs(sin(z*.7-iTime))*.001)*.8;
    }
    O*=vec4(p+.8,0)*z/7e4;
}
