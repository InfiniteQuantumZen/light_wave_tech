#define hash(p )     ( 2.* fract(sin((p)*mat2(127.1,311.7, 269.5,183.3)) *43758.5453123) - 1. ) \
                    *  mat2(cos(iTime+vec4(0,11,33,0)))
#define draw(v,d,w)  clamp(1. - ( abs( fract(v-.5) - .5 ) -.1*(w-1.) ) / (d), 0.,1.)  // method 2: strict
#define hue(v)       ( .6 + .6 * cos( 6.3*(v)  + vec4(0,23,21,0)  ) )
#define Mix(c0,c1,v)     mix(c0,c1,v)

// Automatically toggle between perlin and standard noise over time instead of using keyboard
#define func( P )  ( mod(iTime,4.) > 2. ? perlin( .5*(P) ) : noise( P ) )
#define grad(x,y)      dot( hash( i+vec2(x,y) ), f-vec2(x,y) )

float noise( vec2 p )
{
    vec2 i = floor(p), f = fract(p), // u = f*f*(3.-2.*f);              // = smoothstep
                                        u = f*f*f*( 10. +f*(6.*f-15.)); // better with derivatives
    return mix( mix( grad(0,0), grad(1,0), u.x),
                mix( grad(0,1), grad(1,1), u.x), u.y);
}

float perlin( vec2 U ) { // inspired from https://www.shadertoy.com/view/XdXGW8
    float v = 0., s = .5;
    mat2 m = mat2( 1.6,  1.2, -1.2,  1.6 );
    for( int i=0; i<3; i++, s/=2., U*=m )
    v  += s * noise( U ); 
    return v;
}

#define pi 3.14159
#define oz vec2(1,0)

// random [0,1]
float hash12(vec2 p) {
	vec3 p3  = fract(vec3(p.xyx) * .1031);
	p3 += dot(p3, p3.yzx + 33.33);
	return fract((p3.x + p3.y) * p3.z);
}

// random normalized vector
vec2 randVec(vec2 p) {
	float a = hash12(p);
	a *= 2.0*pi;
	a += iTime;
	return vec2(sin(a), cos(a));
}

// perlin noise
float perlin_bioscanner(vec2 p) {
	vec2 f = fract(p);
	vec2 s = smoothstep(0.0, 1.0, f);
	vec2 i = floor(p);
	float a = dot(randVec(i), f);
	float b = dot(randVec(i+oz.xy), f-oz.xy);
	float c = dot(randVec(i+oz.yx), f-oz.yx);
	float d = dot(randVec(i+oz.xx), f-oz.xx);
	return mix(mix(a, b, s.x), mix(c, d, s.x), s.y)/0.707107;
}

// fractal noise
float fbm(vec2 p) {
	float a = 0.5;
	float r = 0.0;
	for (int i = 0; i < 8; i++) {
		r += a*perlin_bioscanner(p);
		a *= 0.5;
		p *= 2.0;
	}
	return r;
}

// 2D square sdf
float square(vec2 p, vec2 s) {
	vec2 d = abs(p)-s;
	return length(max(d, 0.)) + min(0., max(d.x,d.y));
}

float rand(float seed) {
// This is the classic GLSL 1D random generator
return fract(sin(seed) * 43758.5453123);
} 

void mainImage(out vec4 fragColor, vec2 fragCoord)
{
    vec2 R = iResolution.xy, eps = vec2(1e-3,0),
         U = fragCoord / R.y,
         P = 4.*fragCoord/360. - vec2(0.,.5*iTime);	
    fragColor-=fragColor;
	float l, dl, f = func(P), df, v,a;
    #if 0
        l = exp2(floor(log2(2.*fwidth(P.x)/fwidth(f))));               // subvid amount (relative)
        dl =     fract(log2(2.*fwidth(P.x)/fwidth(f)));      
    #else
        l = exp2(floor(log2(.5/22./fwidth(f))));                       // subvid amount (absolute)
        dl =     fract(log2(.5/22./fwidth(f)));
    #endif

    df = fwidth(f);
    vec4 T = texture(iChannel0, fragCoord/R );
    l *= 12.;
    # if 1
        v = 4.*T.x*T.x+.1;
    # if 0
     fragColor += Mix( draw(    f*l,    l*df , v ),
               draw( 2.*f*l, 2.*l*df , v ),
               dl );
#    else                                          // AA: replace too thin ribbons by intensity
     fragColor += Mix( draw(    f*l,    l*df , max(v,1.) ),
               draw( 2.*f*l, 2.*l*df , max(v,1.) ),
               dl ) * min(v,1.);
#    endif

# else                                                             // color on different layers
#   define Draw(c,ofs)                                    \
      v = 3.*T.c;                                         \
      fragColor.c = mix( draw(    f*l-ofs,    l*df, max(v,1.) ),  \
                 draw( 2.*f*l-ofs, 2.*l*df, max(v,1.) ),  \
                 dl ) * min(v,1.)
    float ofs = .33;   // 0. : too subtle to see.
    Draw(r, 0.);
    Draw(g, ofs);
    Draw(b, -ofs);
# endif
    
    //fragColor = sqrt(fragColor); 


    vec2 uv = fragCoord.xy / iResolution.xy;

    R = iResolution.xy;
    // Normalized coordinates (0.0 to 1.0)
    vec2 U_uv = fragCoord / R; 

    // 1. Create a timer. (e.g., changes twice a second. Change 2.0 to whatever speed you want)
    float flipTimer = floor(iTime * 1.0); 
    
    // 2. Get a random float between 0.0 and 1.0
    float rawRandom = rand(flipTimer);
    
    // 3. Convert it to exactly 0.0 or 1.0 
    // step(0.5, x) returns 0.0 if x is less than 0.5, and 1.0 if x is greater.
    float randomFlip = step(0.5, rawRandom);
    
    if (U_uv.x > .498 && U_uv.x < .500) {
        // Draw white separator
        fragColor = vec4(1.0, 1.0, 1.0, 1.0);
    } 
    else if ( (U_uv.x > .5 && randomFlip == 0.0) || (U_uv.x < .498 && randomFlip == 1.0) ) {
        // Show orig video
        //T = texture(iChannel0, uv);
        //fragColor = T;

        fragColor = sqrt(fragColor);
        vec4 coreColor = texture(iChannel0, uv);
        fragColor.rgb = mix(fragColor.rgb, coreColor.rgb, 0.8);
    } 
    else {
        fragColor = sqrt(fragColor); 
    }




	// coordinates
	uv = fragCoord.xy/iResolution.y;
	uv.y = 1.-uv.y;
	float ar = iResolution.x/iResolution.y;
	uv -= vec2(ar/2.0,0.5);
	float uxy = uv.x+uv.y;

	// distortion noise
	float n = fbm(vec2(10.*uv.y, 3.*iTime));
	float g = 0.2*n-1.9+sin(0.3*iTime);
	uv.x += smoothstep(0.15, 0., abs(g))*n;

	// wave
	float s = perlin_bioscanner(vec2(0.3*iTime));
	vec2 p = vec2(perlin_bioscanner(vec2(0.5*iTime)), 0.5*perlin_bioscanner(vec2(-iTime)))*0.5*(iResolution.xy - vec2(300, 300))/iResolution.y;
	p.y -= 0.1*sin(6.*p.x+iTime);
    vec2 m = (vec2(0.0/iResolution.x, 1.-0.0/iResolution.y)-0.5);
    m.x = clamp(ar*m.x, -ar*0.35, ar*0.35);
    m.y = clamp(m.y, -0.25, 0.25);
    p = p;
	a = smoothstep(-0.5, -0.1, s)*smoothstep(mix(0.01, 0.9, smoothstep(-0.5, 0.5, s)), 0., length(uv-p));
    float r = 0.;
	for (float i = 1.; i <= 8.; i++) {
		float f = uv.y;
		f += 0.2*sin(5.*uv.x+iTime+a*sin(i+iTime)*1.25*cos(5.*uv.x+0.5*i));
		r += 0.003/abs(f);
	}

	// ui
	float ob = square(uv, 0.5*vec2(0.9*ar,1.-0.1*ar));
	float ui = 1.5*smoothstep(fwidth(uxy), -fwidth(uxy), abs(ob));
	float ms = step(ob, -0.005) * smoothstep(0.01, 0.005, abs(abs(uv.y)-0.475*(1.-0.1*ar)));
	ms *= smoothstep(2.*fwidth(uv.x), fwidth(uv.x), fract(uv.x*100.)/100.);
	ui = max(ui, ms);
	float cr = smoothstep(fwidth(uxy), 0., abs(min(abs(uv.x-p.x), abs(uv.y-p.y))));
	cr = max(cr, 3.*smoothstep(fwidth(uxy), -fwidth(uxy), abs(square(uv-p, (0.1+smoothstep(-0.25, 0.5, s))*vec2(0.15)))));
	cr *= smoothstep(0., -0.1, ob);
	ui = max(ui, cr);
	r = max(r, 2.*ui);
	r *= 2.*pow(0.5+0.5*n, 2.);
	r *= 0.5+0.5*hash12(fragCoord.xy+iTime);
	r = clamp(r, 0., 1.);
	vec4 finalColor = vec4(mix(r, pow(1.-r, 3.), smoothstep(-0.025, 0.025, g)));

        fragColor.rgb = mix(fragColor.rgb, finalColor.rgb, finalColor.a);



    // Calculate the intensity of the UI action (how bright the lines/glitches should be)
    float uiIntensity = mix(r, pow(1.-r, 3.), smoothstep(-0.025, 0.025, g));
    
    // Generate a dynamic rainbow color that shifts based on time and UV coordinates
    vec3 rainbowColor = 0.5 + 0.5 * cos(iTime * 3.0 + (uv.x + uv.y) * 4.0 + vec3(0.0, 2.0, 4.0));

    fragColor.rgb = mix(fragColor.rgb, rainbowColor, uiIntensity);

}
