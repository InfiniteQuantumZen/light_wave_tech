// ============================================================
//  Fractal 3D — Infinite Interior Voyage
//  Exported from VJ engine — Shadertoy compatible
//
//  Sliders (tweak these):
// ============================================================
#define ITERATIONS    5.0   // 1–10  : fractal depth
#define SCALE         1.0   // 0.1–5 : structure scale
#define VOYAGE_SPEED  0.5   // 0–2   : forward camera speed
#define TORSION_SPEED 0.5   // 0–2   : rotation twist speed
#define TORSION_OFF   0.0   // offset de phase de torsion
#define DISPERSION    0.5   // 0.01–2: fold offset
#define SHAPE_MORPH   0.0   // 0=cube, 1=sphere
#define TUNNEL_HOLE   0.5   // rayon du tunnel central
#define GLOW          0.5   // 0–2   : bloom intensity
#define TINT          vec3(0.0, 0.667, 1.0)  // hue 200 (cyan-blue)

#define AUDIO_AMOUNT  0.5

// ============================================================

const float PI = 3.14159265359;

mat2 rot(float a) {
    float s = sin(a), c = cos(a);
    return mat2(c, -s, s, c);
}

vec2 path(float z) {
    return vec2(
        sin(z * 0.1) * 2.0 + cos(z * 0.23) * 1.0,
        cos(z * 0.15) * 1.5 + sin(z * 0.31) * 0.8
    );
}

float map(vec3 p, float audioLow) {
    float time = iTime * VOYAGE_SPEED;
    float safeScale = SCALE;
    if (safeScale > 4.0) safeScale = 4.0 + (safeScale - 4.0) * 0.25;

    vec3 pOrig = p;
    pOrig.xy -= path(pOrig.z);

    p.xy -= path(p.z);
    p = mod(p + 2.0, 4.0) - 2.0;

    float s = 1.0;
    float torsion = iTime * TORSION_SPEED + TORSION_OFF;

    for (int i = 0; i < 10; i++) {
        if (float(i) >= ITERATIONS) break;
        p = abs(p) - DISPERSION;
        p.xy *= rot(torsion * 0.1 + float(i) * 0.2);
        p.yz *= rot(torsion * 0.15);
        float scale = 1.5 + audioLow * AUDIO_AMOUNT;
        p *= scale;
        s *= scale;
        p -= vec3(0.5, 1.0, 0.5) * safeScale;
    }

    float box    = (max(abs(p.x), max(abs(p.y), abs(p.z))) - 1.0) / s;
    float sphere = (length(p) - 1.0) / s;
    float fractalDist = mix(box, sphere, SHAPE_MORPH);

    float hole = length(pOrig.xy) - TUNNEL_HOLE;
    return max(fractalDist, -hole);
}

vec3 getNormal(vec3 p, float audioLow) {
    vec2 e = vec2(0.001, 0.0);
    return normalize(vec3(
        map(p + e.xyy, audioLow) - map(p - e.xyy, audioLow),
        map(p + e.yxy, audioLow) - map(p - e.yxy, audioLow),
        map(p + e.yyx, audioLow) - map(p - e.yyx, audioLow)
    ));
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    vec2 uv = (fragCoord / iResolution.xy) * 2.0 - 1.0;
    uv.x *= iResolution.x / iResolution.y;

    // Mouse X simule l'audio low (0–1)
    float audioLow = 0.0;

    // Caméra voyageant le long du path
    float t = iTime * VOYAGE_SPEED;
    float z = t * 10.0;
    vec3 ro = vec3(path(z), z);
    vec3 target  = vec3(path(z + 1.0), z + 1.0);
    vec3 forward = normalize(target - ro);
    vec3 right   = normalize(cross(vec3(0.0, 1.0, 0.0), forward));
    vec3 up      = cross(forward, right);
    vec3 rd      = normalize(forward + uv.x * right + uv.y * up);

    // Raymarching
    float totalD = 0.0;
    float glow   = 0.0;
    int   steps  = 0;
    float d;

    for (int i = 0; i < 80; i++) {
        vec3 p = ro + rd * totalD;
        d = map(p, audioLow);
        if (d < 0.0) d = 0.0001;
        glow += exp(-max(d, 0.0) * 4.0) * GLOW;
        if (d < 0.0005 || totalD > 40.0) break;
        totalD += d * 0.75;
        steps = i;
    }

    vec3 color = vec3(0.0);

    if (totalD < 30.0) {
        vec3 p = ro + rd * totalD;
        vec3 n = getNormal(p, audioLow);

        float diff = max(0.0, dot(n, normalize(vec3(1.0, 1.0, -1.0))));
        float ao   = 1.0 - float(steps) / 64.0;

        color = TINT * (diff + 0.1) * ao;

        // Spéculaire
        float spec = pow(max(0.0, dot(reflect(rd, n), normalize(vec3(1.0, 1.0, -1.0)))), 16.0);
        color += spec * 0.5;
        // Fresnel
        color += pow(1.0 - max(0.0, dot(n, -rd)), 3.0) * 0.3;
    } else {
        color = TINT * 0.05 * (sin(iTime * 0.5) * 0.5 + 0.5);
    }

    // Bloom
    color += glow * 0.015 * TINT;

    // Fog
    color = mix(color, vec3(0.0), clamp(totalD * 0.02, 0.0, 0.95));

    fragColor = vec4(color, 1.0);
}
