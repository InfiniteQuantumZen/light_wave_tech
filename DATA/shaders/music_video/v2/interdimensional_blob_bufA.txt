
void mainImage(out vec4 o, vec2 u) {
    // 1 & 2. Explicitly initialize loop variables and the ray distance!
    // (Also removed the unused variable 'n')
    float i = 0., d = 0., s = 0.;
    
    // 3. Explicitly initialize the out variable to avoid NaN multiplication bugs
    o = vec4(0.0);
    
    vec3 q, p, r = iResolution,
         a = cos(vec3(2,3,1) + iTime/6.);
         
    // Removed the dangerous 'o*=i' from the loop initialization 
    for(; i++ < 1e2; d += s, o += 2.*(1.+cos(.25*i+vec4(2,1,0,0)))/s) {
        q = p = vec3((u+u-r.xy)/r.y * d, d-3.);
        q = a*dot(a,q) - cross(a,q);
        s = 3.3 + 3.*sin(p.y/2.);
        p = abs(fract(q/s)*s - s*.5);
        s = min(max(p.x, p.y),
                 min(max(p.y, p.z),
                 max(p.x, p.z))) - s/4.; 
        s = max(s, 5. - length(q));
        p = q;
        p += .1*(cos(iTime+dot(cos(iTime+p+cos(.3*p)), p) * p));
        p += cos(iTime+p.yzx)*.6;
        s = .001 + .6*min(abs(s),
                        abs(length(p) - 2.)*.7);
    }
    
    o = tanh(o/4e4);
}
