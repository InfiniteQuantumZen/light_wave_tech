#define T(i) cos( i + iTime  + vec2(0,33) )

void mainImage( out vec4 O,  vec2 u )
{
    vec2 R = iResolution.xy,
         U = ( u+u - R ) / min(R.x,R.y),  // uv
         T =  T();

    float f = 0., i = 0.;
    
    for( ; i++ < 10.;  )
        f += 5.0 / abs( length( U +  T( i*.628 ) / 2. ) - (.4+(iMusicSync0/15.0)) ) / 4e2;

    O = f * abs( vec4(T.yx, T.x+T.y , 1 ) );
}
