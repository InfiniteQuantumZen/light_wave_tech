mat2 rot(float a)
{
    float c = cos(a);
    float s = sin(a);
    return mat2(c, -s, s, c);
}

float mapField(vec3 p, float time, out float trap, out float radius)
{
    p.xy = rot(time * 0.25) * p.xy;
    p.x = abs(p.x) - 0.70;
    p.y = abs(p.y) - 0.35;

    trap = 50.0;
    float scale = 1.0;

    for (float j = 0.0; j < 5.0; j += 1.0)
    {
        p = abs(p) / clamp(dot(p, p), 0.18, 2.50) - vec3(0.78, 0.62, 0.42);
        p.yz = rot(0.80 + time * 0.30 + j * 0.18) * p.yz;
        p.xz = rot(0.45 - time * 0.20 - j * 0.11) * p.xz;

        trap = min(trap, min(length(p.xy), length(p.yz)));
        scale *= 1.38;
    }

    radius = length(p);

    float shell = abs(radius - 0.85) / scale;
    float ribs = trap / scale;

    return 0.018 + min(shell, ribs) * 0.55;
}

void mainImage(out vec4 fragColor, vec2 fragCoord)
{
    vec2 uv = (fragCoord * 2.0 - iResolution.xy) / iResolution.y;

    float localTime = mod(iTime, 6.0);
    float time = localTime * 0.25;

//    float time = iTime * 0.25;

    vec3 ro = vec3(0.0, 0.0, -3.80);
    vec3 rd = normalize(vec3(uv, 1.60));

    ro.xz = rot(time * 0.35) * ro.xz;
    rd.xz = rot(time * 0.35) * rd.xz;
    ro.yz = rot(sin(time * 0.50) * 0.25) * ro.yz;
    rd.yz = rot(sin(time * 0.50) * 0.25) * rd.yz;

    vec3 sum = vec3(0.0);
    float t = 0.0;
    float sparkle = 0.0;

    for (float i = 0.0; i < 96.0; i += 1.0)
    {
        vec3 p = ro + rd * t;

        float trap;
        float radius;
        float d = mapField(p, time, trap, radius);

        vec3 palette = 0.5 + 1.5 * cos(
            log(radius + 0.5) * 8.0
            - time * 1.4
            + vec3(0.0, 1.4, 2.8)
        );

        float fog = exp(-0.445 * t * t);
        float pulse = 0.6 + 0.9 * sin(t * 3.5 - time * 0.1 + trap * 2.0);

        sum += palette * pulse * fog / (0.020 + d * d * 140.0);
        sparkle += fog / (1.0 + trap * trap * 120.0);

        t += d;
    }

    vec3 col = sum * sum * 0.00003;
    col += sparkle * vec3(0.08, 0.16, 0.30);

    float vignette = 1.0 - dot(uv, uv) * 0.15;
    col *= vignette;

    col = tanh(col*0.5);

    fragColor = vec4(col, 1.0);
}
