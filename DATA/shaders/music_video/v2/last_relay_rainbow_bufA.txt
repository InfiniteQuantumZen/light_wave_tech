// Modified for Rainbow Colors

const float TIME_SCALE = 0.5;

float hash11(float p)
{
    p = fract(p * 0.1031);
    p *= p + 33.33;
    p *= p + p;
    return fract(p);
}

float hash21(vec2 p)
{
    vec3 p3 = fract(vec3(p.xyx) * 0.1031);
    p3 += dot(p3, p3.yzx + 33.33);
    return fract((p3.x + p3.y) * p3.z);
}

// Helper function to generate rainbow colors based on a single value
vec3 getRainbow(float t) 
{
    // A standard HSV-to-RGB hue calculation
    vec3 p = abs(fract(t + vec3(1.0, 2.0 / 3.0, 1.0 / 3.0)) * 6.0 - 3.0);
    return clamp(p - 1.0, 0.0, 1.0);
}

// Procedural encrypted glyph rain.
vec3 rainLayer(vec2 uv, float scale, float speed, float seed, float t)
{
    vec2 q = uv * vec2(scale, scale * 0.68);
    q.y += t * speed;

    vec2 cell = floor(q);
    vec2 f = fract(q) - 0.5;

    float colRand = hash11(cell.x + seed * 17.13);
    float streamLen = floor(mix(10.0, 30.0, colRand));

    float head = mod(t * speed * 8.0 + colRand * streamLen, streamLen);
    float row  = mod(cell.y + streamLen * 10.0, streamLen);
    float dHead = mod(head - row + streamLen, streamLen);

    float trail = exp(-0.23 * dHead);
    trail *= 1.0 - smoothstep(streamLen * 0.65, streamLen * 0.95, dHead);

    vec2 glyphGrid = vec2(3.0, 5.0);
    vec2 gc = floor((f + 0.5) * glyphGrid);
    vec2 guv = fract((f + 0.5) * glyphGrid) - 0.5;

    float pixel = 1.0 - smoothstep(0.28, 0.42, max(abs(guv.x), abs(guv.y)));
    float bit = step(0.5, hash21(cell * vec2(7.31, 11.17) + gc + seed));
    float bounds = step(abs(f.x), 0.37) * step(abs(f.y), 0.45);

    float flicker = 0.65 + 0.35 * hash21(cell + floor(t * 8.0) + seed);

    float glyph = pixel * bit * bounds * trail * flicker;
    float headFlash = exp(-2.4 * dHead) * glyph;

    // --- COLOR LOGIC UPDATE ---
    // Create a dynamic hue offset driven by screen position, column (cell.x), and time
    float hue = uv.x * 0.15 + cell.x * 0.02 + t * 0.15 + seed * 0.1;
    vec3 baseColor = getRainbow(hue);
    
    // Core glyph uses the rainbow color, slightly brightened
    vec3 glyphColor = baseColor;
    
    // The flashing head mixes the current rainbow hue with white for intensity
    vec3 flashColor = mix(baseColor, vec3(1.0), 0.7);

    return glyph * glyphColor + headFlash * flashColor * 1.8;
}

// Sparse drifting particles.
vec3 particles(vec2 uv, float t)
{
    vec3 color = vec3(0.0);

    vec2 g = uv * 22.0;
    vec2 id = floor(g);
    vec2 f = fract(g) - 0.5;

    for (int j = -1; j <= 1; j++)
    {
        for (int i = -1; i <= 1; i++)
        {
            vec2 offs = vec2(float(i), float(j));
            vec2 cell = id + offs;

            float rnd = hash21(cell + 19.7);

            vec2 basePos = vec2(
                hash21(cell + 1.3),
                hash21(cell + 8.1)
            ) - 0.5;

            vec2 dir = vec2(
                hash21(cell + 4.7),
                hash21(cell + 9.3)
            ) * 2.0 - 1.0;

            dir = normalize(dir + vec2(1e-4));

            float phase = fract(t * (0.03 + rnd * 0.05) + rnd);
            vec2 pos = basePos + dir * ((phase - 0.5) * 0.70);

            vec2 d = f - offs - pos;
            float dist = length(d);

            float spark = exp(-dist * 10.0) * (0.25 + 0.75 * rnd);

            // --- COLOR LOGIC UPDATE ---
            float hue = cell.x * 0.02 + cell.y * 0.02 + t * 0.1 + rnd;
            vec3 c = getRainbow(hue);

            color += c * spark * 0.1;
        }
    }

    return color;
}

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    float t = iTime * TIME_SCALE;

    vec2 uv = (2.0 * fragCoord - iResolution.xy) / iResolution.y;

    vec2 mouse = vec2(0.0);

    vec2 p = uv + mouse * 0.02;

    // --- BACKGROUND COLOR UPDATE ---
    // Changed deep blueish background gradients to neutral dark greys 
    // so the rainbow isn't tinted blue.
    vec3 color = vec3(0.008, 0.008, 0.008);

    // Gentle center ambience
    float radial = exp(-length(p) * 1.35);
    color += vec3(0.030) * radial * 0.60;

    // Mild vertical framing gradient.
    color += vec3(0.012) * smoothstep(1.2, -0.8, uv.y) * 0.6;

    // Fine display grain.
    float grain = hash21(floor(fragCoord * 0.5) + floor(t * 15.0));
    color += vec3(0.015) * grain * 0.18;

    // Sparse particle layer.
    color += particles(p, t);

    // Encrypted glyph rain.
    color += rainLayer(p + vec2(0.10, 0.0), 14.0, 0.70, 3.0, t) * 0.22;
    color += rainLayer(p * 1.10 - vec2(0.18, 0.0), 20.0, 1.00, 17.0, t) * 0.16;
    color += rainLayer(p * 1.24 + vec2(0.27, 0.0), 28.0, 1.28, 41.0, t) * 0.10;

    // Very subtle moving horizontal scan energy (Also tinted rainbow!)
    float scanBand1 = exp(-90.0  * abs(fract(uv.y * 20.0 - t * 0.42) - 0.5));
    float scanBand2 = exp(-120.0 * abs(fract(uv.y * 31.0 + t * 0.27) - 0.5));

    color += getRainbow(uv.y * 0.5 + t * 0.3) * scanBand1 * 0.02;
    color += getRainbow(uv.y * 0.7 - t * 0.2) * scanBand2 * 0.01;

    // Fine scanlines, no sine.
//    float scanlines = smoothstep(0.18, 0.82, fract(fragCoord.y * 0.5));
//    color *= 0.978 + 0.022 * scanlines;

    // Soft vignette.
//    float vignette = 1.0 - 0.34 * smoothstep(0.3, 1.6, length(uv));
//    color *= vignette;

    // Tonemap / polish.
//    color = 1.0 - exp(-color * 1.28);
    // Removed the cyan tint multiplier `color *= vec3(0.92, 1.03, 1.05);` 
    // to preserve accurate rainbow colors.
//    color = pow(color, vec3(0.94));

    fragColor = vec4(color, 1.0);
}
