vec2 hash22(vec2 p) {
p = vec2(dot(p, vec2(127.1, 311.7)), dot(p, vec2(269.5, 183.3)));
return -1.0 + 2.0 * fract(sin(p) * 43758.5453);
}

float perlinNoise(vec2 p) {
vec2 i = floor(p);
vec2 f = fract(p);
vec2 u = f * f * (3.0 - 2.0 * f);
return mix(mix(dot(hash22(i), f),
dot(hash22(i + vec2(1, 0)), f - vec2(1, 0)), u.x),
mix(dot(hash22(i + vec2(0, 1)), f - vec2(0, 1)),
dot(hash22(i + vec2(1, 1)), f - vec2(1, 1)), u.x), u.y);
}
float hash21(vec2 p) {
p = fract(p * vec2(0.3183099, 0.3678794) + 0.1);
p *= 17.0;
return fract(p.x * p.y * (p.x + p.y));
}

float valueNoise(vec2 p) {
vec2 i = floor(p);
vec2 f = fract(p);
f = f * f * (3.0 - 2.0 * f);
return mix(mix(hash21(i), hash21(i + vec2(1, 0)), f.x),
mix(hash21(i + vec2(0, 1)), hash21(i + vec2(1, 1)), f.x), f.y);
}

float fbm(vec2 p) {
float value = 0.0;
float amplitude = 0.5;
for (int i = 0; i < 5; i++) {
value += amplitude * valueNoise(p);
p *= 2.0;
amplitude *= 0.5;
}
return value;
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
vec2 uv = fragCoord.xy / iResolution.y;
float t = iTime*0.5;
float n = perlinNoise(uv * 4.0 + t);
n = n * 0.5 + 0.5;
float n1 = perlinNoise(uv + n*2.0+t*1.1415926535);
float n2 = perlinNoise(uv + n1*10.0+-t);
float n3 = fbm(uv+n1*10.0+t+0.5);
float n4 = mix(n2,n3,0.5);
vec3 col = vec3(0.0);
     col =mix(vec3(0.1,0.0,0.2),vec3(1.0,0.6,1.0),n4);
     col = smoothstep(0.15, 0.5, col);
fragColor = vec4(col, 1.0);
}
