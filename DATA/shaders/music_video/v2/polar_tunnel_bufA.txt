// ichannel1: pebbles_512x512.png

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // Normalized pixel coordinates (from 0 to 1)
    vec2 uv = (2.0 * fragCoord - iResolution.xy) / iResolution.y;

    float theta = atan(uv.y, uv.x);
    float d = length(uv);

    float z = 1.0 / d;
    //d *= .3;
    
    vec3 p = vec3(0,0,z);
    
    z += iTime * .2;
    
    float displacement = textureLod(iChannel1, vec2(theta / 3.1415926 / 2.0 + 0.5, iTime*.02), 1.0).x;
    z -= displacement * 2.;

    // Time varying pixel color
    float f = sin((z + iTime * 1.)*5.) * .5 + .5;
    
    vec3 col = f + vec3(1) * smoothstep(-1.0, 1.0, sin(z * 100.0 + theta + 2.0*vec3(.2,.6,1.)));
   
    col *= exp(-p.z*.7);
    col *= displacement;
    
    col = smoothstep(0.0, 0.6, col);

    // Output to screen
    fragColor = vec4(col,1.0);
}
