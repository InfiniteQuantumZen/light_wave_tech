float det=.001,t, boxhit;
vec3 adv, boxp;

float hash(vec2 p)
{
	vec3 p3  = fract(vec3(p.xyx) * .1031);
    p3 += dot(p3, p3.yzx + 33.33);
    return fract((p3.x + p3.y) * p3.z);
}


mat2 rot(float a)
{
    float s=sin(a), c=cos(a);
    return mat2(c,s,-s,c);
}

vec3 path(float t)
    {
    vec3 p=vec3(vec2(sin(t*.1),cos(t*.05))*10.,t);
    p.x+=smoothstep(.0,.5,abs(.5-fract(t*.02)))*10.;
    return p;
}

float fractal(vec2 p)
{
    p=abs(5.-mod(p*.2,10.))-5.;
    float ot=1000.;
    for (int i=0; i<7; i++)
    {
        p=abs(p)/clamp(p.x*p.y,.25,2.)-1.;
        if(i>0)ot=min(ot,abs(p.x)+.7*fract(abs(p.y)*.05+t*.05+float(i)*.3));
        
    }
    ot=exp(-10.*ot);
    return ot;
}

float box(vec3 p, vec3 l)
{
    vec3 c=abs(p)-l;
    return length(max(vec3(0.),c))+min(0.,max(c.x,max(c.y,c.z)));
}

float de(vec3 p)
{
    boxhit=0.;
    vec3 p2=p-adv;
    p2.xz*=rot(t*.2);
    p2.xy*=rot(t*.1);
    p2.yz*=rot(t*.15);
    float b=box(p2,vec3(1.));
    p.xy-=path(p.z).xy;
    float s=sign(p.y);
    p.y=-abs(p.y)-3.;
    p.z=mod(p.z,20.)-10.;
    for (int i=0; i<5; i++)
    {
        p=abs(p)-1.;
        p.xz*=rot(radians(s*-45.));
        p.yz*=rot(radians(90.));
    }
    float f=-box(p,vec3(5.,5.,10.));
    float d=min(f,b);
    if (d==b) boxp=p2, boxhit=1.;
    return d*.7;
}


vec3 march(vec3 from, vec3 dir)
{
    vec3 p,n,g=vec3(0.);
    float d, td=0.;
    for (int i=0; i<80; i++)
    {
        p=from+td*dir;
        d=de(p)*(1.-hash(gl_FragCoord.xy+t)*.3);
        if (d<det && boxhit<.5) break;
        td+=max(det,abs(d));
        float f=fractal(p.xy)+fractal(p.xz)+fractal(p.yz);
        //boxp*=.8-iMusicSync0;
        float b=fractal(boxp.xy)+fractal(boxp.xz)+fractal(boxp.yz);
        //vec3 colf=vec3(iMusicSync2+(f/1.2),f*(f/1.2),f*(b/2.0));
        // ORIG vec3 colf=vec3(f*f,f,f*f*b);
        //vec3 colb=vec3(f*f+.1,iMusicSync2+b+.03,f+0.01);        
        // ORIG vec3 colb=vec3(b+.1,b*b+.05,0.);

        if(iMusicSync0 > 0.3) {
            // Cyan Tunnel: Low Red (f*f*f), High Green (f), High Blue (f)
            vec3 colf = vec3(f*f*f, f, f);        
            // Magenta Cube: High Red (b+.1), Low Green (b*b*0.1), High Blue (b+.11)
            vec3 colb = vec3(b + 0.05, b*b*0.04, b + 0.08);
            g+=colf/(3.+d*d*2.)*exp(-.0015*td*td)*step(5.,td)/2.*(1.-boxhit);
            g+=colb/(10.+d*d*20.)*boxhit*.5;
        }
        else if(iMusicSync0 > 0.5) {
            vec3 colf=vec3(f*f,f,f*f*b);
            vec3 colb=vec3(b+.1,b*b+.05,0.);
            g+=colf/(3.+d*d*2.)*exp(-.0015*td*td)*step(5.,td)/2.*(1.-boxhit);
            g+=colb/(10.+d*d*20.)*boxhit*.5;
        }
        else {
            // Magenta Tunnel: High Red (f), Low Green (f*f*f), High Blue (f)
            vec3 colf = vec3(f, f*f*f, f);    
            // Cyan Cube: Low Red (b*b*0.0.07), High Green (b+.0.07), High Blue (b+.11)
            vec3 colb = vec3(b*b*0.07, b + 0.07, b + 0.11);
            g+=colf/(3.+d*d*2.)*exp(-.0015*td*td)*step(5.,td)/2.*(1.-boxhit);
            g+=colb/(10.+d*d*20.)*boxhit*.5;
        }
    }
    return g;
}

mat3 lookat(vec3 dir, vec3 up) 
{
    dir=normalize(dir);vec3 rt=normalize(cross(dir,normalize(up)));
    return mat3(rt,cross(rt,dir),dir);
}

// 1. Put this function ABOVE your mainImage function
float rand(float seed) {
    // This is the classic GLSL 1D random generator
    return fract(sin(seed) * 43758.5453123);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = (fragCoord-iResolution.xy*.5)/iResolution.y;
    t=iTime*7.;
    vec3 from=path(t);
    adv=path(t+6.+sin(t*.1)*3.);
    vec3 dir=normalize(vec3(uv,.7));
    dir=lookat(adv-from,vec3(0.,1.,0.))*dir;
    vec3 col=march(from, dir);
    fragColor=vec4(col,1.0);


    // MUSIC_VIDEO__MOD
    float colorTimer = floor(iTime * 2.0); 

    vec3 flashColor = vec3(
        rand(colorTimer),           
        rand(colorTimer + 10.0),    
        rand(colorTimer + 20.0)     
    );

    // Add the flash to the transformed screen
    vec3 finalColor = fragColor.rgb + (flashColor * iMusicSync2*1.2);

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 

    // --- UNIVERSAL FLASH TO WHITE (FIXED TIMING) ---
    // 1. Define exactly how long this animation should take (e.g., 2.5 seconds)
    float transitionTime = 1.0; 

    // 2. Create a timer that stays at 0.0 for the whole video, 
    // but starts counting up exactly 'transitionTime' seconds before the end.
    float triggerTime = iEndTime - transitionTime;
    float tFlash = clamp(iTime - triggerTime, 0.0, transitionTime);

    // 3. Run your exact same math, but scale it to 'transitionTime' instead of 'duration'
    float z_prog = smoothstep(0.0, transitionTime, tFlash);
    z_prog = pow(z_prog, 1.5); // Easing for accelerating zoom effect
    
    float animZ = mix(20.0, -2.0, z_prog);
    float flashAmt = 1.0 - smoothstep(-1.0, 1.0, animZ);
    
    // 4. Mix and output
    vec3 colFlash = mix(fragColor.rgb, vec3(1.0), flashAmt);
    fragColor = vec4(colFlash, 1.0);
}
