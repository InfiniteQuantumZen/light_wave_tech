// A pure zig-zag triangle wave returning -1.0 to 1.0
float zig(float x) {
    return abs(fract(x) - 0.5) * 4.0 - 1.0;
}

// 2D zig-zag for translations and paths
vec2 zig2(float x, float offset) {
    return vec2(zig(x), zig(x + offset));
}

float sdBox(vec3 p, vec3 b) {
    vec3 d = abs(p) - b;
    return min(max(d.x, max(d.y, d.z)), 0.0) + length(max(d, 0.0));
}

// Palette driven by a zig-zag triangle wave
vec3 palette(float t) {
    vec3 a = vec3(0.5, 0.5, 0.5);
    vec3 b = vec3(0.5, 0.5, 0.5);
    vec3 c = vec3(1.0, 1.0, 1.0);
    vec3 d = vec3(0.263, 0.416, 0.557);
    
    vec3 wave = abs(fract(c * t + d) - 0.5) * 4.0 - 1.0;
    return a + b * wave;
}

float map(vec3 p) {
    // Zig-zag path translation (amplitude reduced so it doesn't crash into walls)
    p.xy -= zig2(p.z * 0.1 + iTime * 0.2, 0.25) * 0.75;
    
    // Linearly pulse the scale
    float pulse = abs(fract(p.z * 0.5 - iTime * 0.5) - 0.5) * 0.2;
    p.xy *= 1.0 + pulse;
    
    // Cubic grid repetition (Spaced out much further now!)
    p.z = mod(p.z, 1.0) - 0.5;
    p.xy = mod(p.xy, 1.0) - 0.5;
    
    // The main cube (slightly larger but within a much larger grid space)
    float box = sdBox(p, vec3(0.25));
    
    // Subtracting elongated boxes to create hollow square holes
    float h1 = sdBox(p, vec3(0.3, 0.1, 0.1));
    float h2 = sdBox(p, vec3(0.1, 0.3, 0.1));
    float h3 = sdBox(p, vec3(0.1, 0.1, 0.3));
    float holes = min(h1, min(h2, h3));
    
    float shape = max(box, -holes);
    return shape - 0.005; // slight bevel
}

vec3 calcNormal(vec3 p) {
    vec2 e = vec2(0.002, 0.0);
    return normalize(vec3(
        map(p + e.xyy) - map(p - e.xyy),
        map(p + e.yxy) - map(p - e.yxy),
        map(p + e.yyx) - map(p - e.yyx)
    ));
}

void mainImage(out vec4 fragColor, vec2 fragCoord) {
    vec2 uv = (fragCoord - 0.5 * iResolution.xy) / iResolution.y;
    
    // Setup camera origin and path
    vec3 ro = vec3(0.0, 0.0, -iTime * 1.2);
    ro.xy += zig2(ro.z * 0.1 + iTime * 0.2, 0.25) * 0.75;
    
    // Look along the zig-zag path to stay inside the tunnel
    vec3 lookAt = vec3(0.0, 0.0, ro.z + 1.0);
    lookAt.xy += zig2(lookAt.z * 0.1 + iTime * 0.2, 0.25) * 0.75;
    
    vec3 forward = normalize(lookAt - ro);
    vec3 right = normalize(cross(vec3(0.0, 1.0, 0.0), forward));
    vec3 up = cross(forward, right);
    
    // Widened Field of View (FOV)
    float fov = 0.6; 
    vec3 rd = normalize(forward * fov + uv.x * right + uv.y * up);
    
    float dist = 0.0;
    vec3 color = vec3(0.0);
    float glow = 0.0;
    
    for(int i = 0; i < 90; i++) {
        vec3 p = ro + rd * dist;
        float d = map(p);
        
        glow += 0.0015 / (0.0005 + abs(d));
        
        if(d < 0.001) {
            vec3 n = calcNormal(p);
            
            float fresnel = pow(1.0 - max(dot(n, -rd), 0.0), 2.5);
            vec3 albedo = palette(p.z * 1.5 + iTime * 0.5 + p.x);
            
            color = albedo * fresnel * 2.5;
            
            // Zig-zagging light direction
            vec3 lightDir = normalize(vec3(zig(iTime * 0.2), 1.0, zig(iTime * 0.2 + 0.25)));
            float spec = pow(max(dot(reflect(rd, n), lightDir), 0.0), 32.0);
            color += spec * albedo * 2.0;
            
            break;
        }
        
        if(dist > 25.0) break; // Extended render distance
        dist += d * 0.6; 
    }
    
    vec3 glowColor = palette(dist * 0.2 - iTime) * glow * 0.02;
    color += glowColor;
    
    color = mix(color, vec3(0.02, 0.0, 0.05), 1.0 - exp(-0.005 * dist * dist));
    
    color = smoothstep(0.0, 1.0, color); 
    color = pow(color, vec3(1.0 / 2.2));
    

    float vignette = length(uv);
    color.r *= 1.0 + vignette * 0.05 * zig(iTime * 2.0);
    color.b *= 1.0 - vignette * 0.05 * zig(iTime * 2.5 + 0.5);
    
    fragColor = vec4(color, 1.0);
}
