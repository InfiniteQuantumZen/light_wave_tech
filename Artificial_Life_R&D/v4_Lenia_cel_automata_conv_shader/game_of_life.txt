// game_of_life.txt

void mainImage( out vec4 fragColor, in vec2 fragCoord ) {
    vec2 uv = fragCoord / iResolution.xy;
    
    // Read the PyTorch Tensor!
    vec4 state = texture(iChannel0, uv);

    float alive = state.r;     // Main Lenia Density [0, 1]
    float history = state.g;   // Fade Momentum [0, 1]

    vec3 bgColor = vec3(0.01, 0.02, 0.06);     
    vec3 shellColor = vec3(0.1, 0.5, 0.7);     
    vec3 coreColor = vec3(1.0, 0.9, 0.2);      

    vec3 col = bgColor;
    
    col = mix(col, vec3(0.0, 0.1, 0.3), history);
    vec3 fluidColor = mix(shellColor, coreColor, smoothstep(0.4, 0.9, alive));
    col = mix(col, fluidColor, smoothstep(0.1, 0.5, alive));

    vec2 p = -1.0 + 2.0 * uv;
    float r = sqrt(dot(p, p));
    col *= 1.0 - 0.2 * r;

    fragColor = vec4(col, 1.0);
}
