void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec4 o = vec4(0.0);
    vec3 p = vec3(0.0);

    float z = 0.0;
    float d = 0.0;

    vec2 r = iResolution.xy;

    for (int i = 0; i < 70; i++)
    {
        p = z * normalize(vec3(fragCoord, 0.0) * 2.0 - r.xyy);
        p.z += iTime / 0.08;

        float rot = iTime * -0.85;

        mat2 R = mat2(
            cos(rot), -sin(rot),
            sin(rot),  cos(rot)
        );

        p.xy = R * p.xy;

        vec2 warped = p.xy + 0.6 * sin(p.yx * 2.7 + p.z * -0.08);

        float angle = atan(warped.y, warped.x);
        float angularTerm = cos(angle * 4.0 + sin(p.z * 0.2) * 1.5);

        vec2 q = abs(warped);
        float radialTerm = pow(pow(q.x, 4.0) + pow(q.y, 4.0), 0.25) - 4.2;

        d = 0.01 + 0.42 * length(vec2(angularTerm + 0.3, radialTerm));

        z += d;

        o += 0.04 * (-cos(z*0.5 + sin(p.z)*-0.4 + vec4(6.0, 1.0, 2.0, 0.0)) + 1.2) / d / z;
    }

    fragColor = tanh(o*0.5);
}
