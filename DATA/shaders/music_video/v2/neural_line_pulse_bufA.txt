#define R iResolution.xy
#define PIXEL 10.0/min(R.x,R.y)
#define S smoothstep
#define T iTime

#define L(p,b) length( p - (b) * clamp( dot(p,b)/dot(b,b), 0., 1. ) )

const float a[3] = float[](0.,1.,-1.);

float hash(vec2 p) {
    p = fract(p * vec2(123.34, 456.21));
    p += dot(p, p + 45.32);
    return fract(p.x * p.y);
}

float cn(vec2 a, vec2 d) {
    float n = hash(a);

    vec2 c = normalize(a), l = normalize(d);
    //Alignment
    float m = abs(dot(c, l)),
    //Probability
    p = S(.7, .98, m)*.75, t = length(a),    
    //Central concentration
    ce = 1.0 - S(0.0, 2., t);
    //The farther the branch, the lower the probability.
    p *= S(18.0, 2.0, t);
    p = mix(p, 0.8, ce);

    return mix(1e2,0.,step(n, p));
}

void mainImage(out vec4 O, in vec2 F) {
    vec2 uv = (F - 0.5 * R) / R.y;

    float s = 20., d = 1e2;
    vec2 u = uv * s,
        i = floor(u),
        f = fract(u) - .5; 
    
    // Eight directions and their axis
    // draw line
    for(int m = 1; m < 9; ++m)
    {
        int j = m%3, k = m/3;
        vec2 q = vec2(a[j], a[k]), p = i + q*.5;
        d = min(d, cn(p, q)+L(f, q*.5));
    }

    //Color
    float t = 0.1, an = atan(i.y, i.x), di = length(uv)*5.,
    //Border
    l = S(t + PIXEL, t, d) - S(t, 0., d),
    //Core fine line
    cl = S(t*.3 + PIXEL, t*.3, d),
    //Glow
    g = PIXEL / (d + 1e-3),
    
    //Random Offset/Speed
    ro = sin(an*7.)*2.5 + cos(an*3.)*1.5,
    rs = 1.5 + sin(an*5.)*.5,
    //Phase
    w = fract(di + ro - T*rs),
    //Pulse
    pu = .08 / (1. - w + .015) * hash(uv*20.);
    pu = max(0., pu - .05) * S(1., .95, w); 
    
    // --- RAINBOW COLOR MODIFICATION ---
    // We create 3 different shifting rainbow gradients based on Time (T).
    // Adding distance (di) and angle (an) makes the rainbow colors swirl across the screen organically.
    vec3 rainbow1 = 0.5 + 0.5 * cos(T + di * 0.5 + an + vec3(0.0, 2.0, 4.0));
    vec3 rainbow2 = 0.5 + 0.5 * cos(T + di * 0.5 + an + vec3(1.0, 2.0, 3.0));
    vec3 rainbow3 = 0.5 + 0.5 * cos(T + di * 0.5 + an + vec3(2.0, 3.0, 4.0));

    // Apply our new rainbow colors to the base border (bc), core (pc), and glow (g)
    vec3 bc = rainbow1 * l * 1.5;
    vec3 pc = rainbow2 * pu * cl * 4.0;
    pc += rainbow3 * pu * g * 1.5;
    // ----------------------------------

    vec3 c = bc + pc;
    c *= exp(-di * .5);

    O = vec4(c, 1.);
}
