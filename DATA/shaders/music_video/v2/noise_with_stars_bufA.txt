// Функция плавного шума (2D Noise)
float noise(in vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    vec2 u = f * f * (3.0 - 2.0 * f);
    
    return mix(mix(fract(sin(dot(i + vec2(0.0,0.0), vec2(127.1,311.7))) * 43758.5453123),
                   fract(sin(dot(i + vec2(1.0,0.0), vec2(127.1,311.7))) * 43758.5453123), u.x),
               mix(fract(sin(dot(i + vec2(0.0,1.0), vec2(127.1,311.7))) * 43758.5453123),
                   fract(sin(dot(i + vec2(1.0,1.0), vec2(127.1,311.7))) * 43758.5453123), u.x), u.y);
}
#define iterations 17
#define formuparam 0.53

#define volsteps 20
#define stepsize 0.1

#define zoom   0.800
#define tile   0.850
#define speed  0.010 

#define brightness 0.0015
#define darkmatter 0.300
#define distfading 0.730
#define saturation 0.850

// Фрактальный шум для турбулентности жидкости
float fbm(in vec2 p) {
    float v = 0.0;
    float a = 0.5;
    vec2 shift = vec2(100.0);
    mat2 rot = mat2(cos(0.5), sin(0.5), -sin(0.5), cos(0.5));
    for (int i = 0; i < 6; ++i) {
        v += a * noise(p);
        p = rot * p * 2.0 + shift;
        a *= 0.5;
    }
    return v;
}

// Функция для создания закручивания (вихря) вокруг центра
vec2 rotate(vec2 p, float a) {
    float c = cos(a);
    float s = sin(a);
    return vec2(p.x * c - p.y * s, p.x * s + p.y * c);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord) {
    // Нормализация координат (центр экрана — это (0.0, 0.0))
    vec2 uv = (fragCoord - 0.5 * iResolution.xy) / iResolution.y;
        vec2 uv4=fragCoord.xy/iResolution.xy-.5;
	uv4.y*=iResolution.y/iResolution.x;
	vec3 dir=normalize(vec3(uv4*zoom,0.015));
   
	float time=iTime*speed+.25;

	float t3 = iTime * .1 + ((.25 + .05 * sin(iTime * .1))/(length(uv.xy) + .27)) * 2.2;
	float si = sin(t3);
	float co = cos(t3);
	mat2 ma = mat2(co, si, -si, co);
     uv.xy*=ma;
	vec3 from=vec3(1.,.5,0.5);

	//volumetric rendering
	float s=0.1,fade=1.;
	vec3 v=vec3(0.);
	for (int r=0; r<volsteps; r++) {
		vec3 p=from+s*dir*.5;
      
		p = abs(vec3(tile)-mod(p,vec3(tile*2.))); // tiling fold
		float pa,a=pa=0.;
		for (int i=0; i<iterations; i++) { 
			p=abs(p)/dot(p,p)-formuparam;
            p.xy*=mat2(cos(iTime*0.03),sin(iTime*0.03),-sin(iTime*0.03),cos(iTime*0.03) );// the magic formula
			a+=abs(length(p)-pa); // absolute sum of average change
			pa=length(p);
		}
		float dm=max(0.,darkmatter-a*a*.001); //dark matter
		a*=a*a; // add contrast
		if (r>6) fade*=1.3-dm; // dark matter, don't render near
		//v+=vec3(dm,dm*.5,0.);
		v+=fade;
		v+=vec3(s,s*s,s*s*s*s)*a*brightness*fade; // coloring based on distance
		fade*=distfading; // distance fading
		s+=stepsize;
	}
	v=mix(vec3(length(v)),v,saturation); //color adjust
    // Дистанция и угол от центра экрана
    float dist = length(uv);
    float angle = atan(uv.y, uv.x);
    
    // Скорость анимации
    float t = iTime * 0.4;
    
    // --- Симуляция направленных всплесков к центру ---
    // Создаем 4 источника флюидов с разных сторон (север, юг, восток, запад)
    // Используем синус от времени, чтобы они «выплевывались» импульсами
    float pulse = sin(t * 1.5) * 0.5 + 0.5;
    
    // Искажаем координаты так, будто жидкость течет снаружи ОКОЛО центра
    // Направление движения: радиальное (снаружи внутрь) + вихрь в самом центре
    float flow = t * 0.5 - (1.0 / (dist + 0.15)) * (0.3 + pulse * 0.2);
    
    // Закручиваем пространство сильнее ближе к центру (эффект водоворота для перемешивания)
    float swirl = 3.1415 * smoothstep(1.2, 0.0, dist) * (sin(t * 0.5) * 0.5 + 1.0);
    vec2 uvFlow = rotate(uv, angle + swirl);
    
    // Переводим в полярно-шумовые координаты для имитации потоков струй
    vec2 p = vec2(angle * 1.5, flow);
    
    // --- Сложное перемешивание (Domain Warping) ---
    // Первый слой: генерирует «рукава» флюидов, вливающиеся извне
    vec2 q = vec2(
        fbm(p + vec2(0.0, t * 0.2)),
        fbm(p + vec2(2.4, -t * 0.3))
    );
    
    // Второй слой: создает мелкомасштабное турбулентное смешивание в эпицентре
    vec2 r = vec2(
        fbm(p + 3.0 * q + vec2(1.7, 9.2) + vec2(0.0, t * 0.5)),
        fbm(p + 3.0 * q + vec2(8.3, 2.8) + vec2(t * 0.2, 0.0))
    );
    
    // Финальная текстура смешивания красок
    float f = fbm(p + 4.0 * r);
    
    // --- Цветовая палитра (OLED Splash) ---
    // Базовый цвет космоса / глубокой темноты
    vec3 color = vec3(0.01, 0.0, 0.03);
    
    // Распределяем цвета в зависимости от направления всплесков (угла) и шума
    // Разные стороны выдают разный цвет флюида
    vec3 fluidColor1 = vec3(1.0, 0.0, 0.4); // Насыщенный розовый
    vec3 fluidColor2 = vec3(0.0, 0.8, 1.0); // Яркий бирюзовый
    vec3 fluidColor3 = vec3(0.5, 0.0, 0.9); // Фиолетовый
    
    // Смешиваем цвета флюидов на основе угла их «выплевывания»
    vec3 mixedFluid = mix(fluidColor1, fluidColor2, sin(angle * 2.0) * 0.5 + 0.5);
    mixedFluid = mix(mixedFluid, fluidColor3, cos(angle * 3.0) * 0.5 + 0.5);
    
    // Проявляем флюиды: они ярче там, где активен шум, и затухают на самых краях экрана
    float edgeMask = smoothstep(1.5, 0.2, dist);
    color = mix(color, mixedFluid, clamp(length(q) * f * 1.8 * edgeMask, 0.0, 1.0));
    
    // Добавляем супер-яркие неоновые прожилки в местах самого жесткого перемешивания (в центре)
    color = mix(color, vec3(1.0, 0.9, 0.5)*v*.05, clamp(length(r.x) * pow(f, 2.0) * smoothstep(0.8, 0.0, dist), 0.0, 1.0));
    
    // Белые глянцевые блики на пиках волн краски
    color += vec3(1.0, 1.0, 1.0) * pow(f, 6.0) * 2.5 * smoothstep(1.0, 0.1, dist);
    
    // Затемнение (виньетка) по самым краям экрана, чтобы акцентировать внимание на центре
    color *= smoothstep(1.4, 0.4, dist);
    
    // Повышение контраста для OLED
    color = smoothstep(0.0, 1.0, color);
    color = pow(color, vec3(1.1));
    
    fragColor = vec4(color, 1.0);
}
