#define rot(x) mat2(cos(x+vec4(0,11,33,0)))

//formula for creating colors;
#define H(h)  (  cos(  h/2. + vec3(1,6,3)   )*.3  + .3 )


#define R iResolution
     
#define ex exp(w.x)
#define cy cos(w.y)
#define sy sin(w.y)
#define cz cos(w.z)
#define sz sin(w.z)
#define cw cos(w.w)
#define sw sin(w.w)


void mainImage( out vec4 O, vec2 U) {
      
    O = vec4(0);
    
    vec3 c=vec3(0);
    vec4 rd = normalize( vec4(U-.5*R.xy, R.y, 0))*10.;
    
    float dotp, totdist=0., tt=iTime, t=0.;
    
    for (float i=0.; i<100.; i++) {
        
        vec4 p = vec4( rd*totdist) ;
        
        float shell = length(p) - 1.;
        p.z  -= -2.;
        p.x += 2.4;
        p.y -= 2.5 + mod(tt/3.,5.);
        
        p.xz *= rot( 1.4*(1.+.2*sin(tt/3.) ) ) ;
        p.xy *= rot(1. + .1*sin(tt/4.)); 
        p.yz *= rot(tt/5.);

        vec4 w=vec4(0) , dz=vec4(0), z;
     
        
        for (float j=0.; j<3.; j++) {

                z = vec4( 
                        ex*(cy*cz*cw - sy*sz*sw),
                        ex*(sy*cz*cw + cy*sz*sw),
                        ex*(cy*sz*cw - sy*cz*sw),
                        ex*(sy*sz*cw + cy*cz*sw)
                        );
                       
                //iterated derivative here instead of just scale factor
                dz = vec4(z.x*dz.x-z.y*dz.y-z.z*dz.z-z.w*dz.w,
                          z.x*dz.y+z.y*dz.x+z.z*dz.w-z.w*dz.z,
                          z.x*dz.z-z.y*dz.w+z.z*dz.x+z.w*dz.y,
                          z.x*dz.w+z.y*dz.z-z.z*dz.y+z.w*dz.x);
                          
                dz += .5;
                
                dotp = 2.6*clamp(1./dot(z.xz,z.xz),.01,7.);
                
                dz *=  dotp;
                
                w = z *dotp + p ;
                        
        }
         
        float ddz = clamp(dot(dz,dz),1e-2,1e4);
        float ddw = clamp(dot(w.xw,w.xw),1e-2,1e4);
        
        float dist = max(-shell, (sqrt(ddw/ddz)) ) ;

        float stepsize = dist/30.;     
        totdist += stepsize;
     
        
        if (i>10.)
        c +=
             .04*
             H( log(ddz) ) * exp(-i*i*dist*dist/1e2)
             + .02*cos(atan(w.w,w.x)/2. + vec3(1,6,3) ) * exp(-i*i*dist*dist/5e2)
             ;
    }
    
    c = 1. - exp(-c*c);
    O = ( vec4(c*c,0) );
    
               
}
