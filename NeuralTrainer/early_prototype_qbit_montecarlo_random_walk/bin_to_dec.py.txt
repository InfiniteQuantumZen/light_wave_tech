def generate_capped_randoms(file_path):
    CEILING = 20000
    with open(file_path, 'r') as file:
        for line in file:
            decimal_str = line.strip()
            if decimal_str:
                decimal = int(decimal_str)
                capped_random = decimal % CEILING  # Derive a number in [0, CEILING-1]
                yield capped_random
