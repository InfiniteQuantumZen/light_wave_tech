void mainImage(out vec4 O,vec2 I){
    float localTime = mod(iTime, 6.0);
    float T=localTime+70.0,t1=T*4.89,t2=T*.53,t3=T*2.43,i=0.,z=0.,d;
    for(O-=O;i++<60.;){
        vec3 p=z*normalize(vec3(I+I-iResolution.xy,iResolution.y)),q;
        p.z-=7.00+sin(T);
        p.yz*=mat2(cos(t1*=i/120.),-sin(t1),sin(t1),cos(t1));
        p.xz*=mat2(cos(t2*=i/120.),-sin(t2),sin(t2),cos(t2));
        q=p;
        for(d=1.;d++<6.;q=abs(q)-.6,q+=sin(q.zxy*(-.3*d-0.021483*i*sin(T))+t3)/d*1.1);
        z+=d=.01+abs(length(q)-1.2+sin(q.x*4.+t3)*cos(q.y*4.-t1)*.75)/3.6;
        O+=(cos(length(q)*1.7+t2+vec4(0,1.2,2.4,0))+1.3)/(d*z*130.);
    }
    O=tanh(O*O*.5);
}
