#define t iTime
#define r iResolution.xy

// 1. Put this function ABOVE your mainImage function
float rand(float seed) {
    // This is the classic GLSL 1D random generator
    return fract(sin(seed) * 43758.5453123);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord ){
	vec3 c;
	float l,z=t;
	for(int i=0;i<3;i++) {
		vec2 uv,p=fragCoord.xy/r;
		uv=p+(iMusicSync0/5.0);
		p-=.5;
		p.x*=r.x/r.y;
		z+=.07;
		l=length(p);
		uv+=p/l*(tan(z)+1.)*abs(tan(l*9.-z*2.));
		c[i]=.01/length(abs(mod(uv,1.)-.5))+(iMusicSync0/4.0);
	}
	fragColor=vec4(c/l,t);
}
