/*
Only 10 Raymarch Steps! Other looks can be found with as few as 5 steps.
This shader was influenced by Xor's Surf 2, but created using a different technique.
Xor's Surf 2: https://www.shadertoy.com/view/3fKSzc
Methods were drawn from my Abstract Series: https://www.shadertoy.com/playlist/D3VBDt
In this series, I aimed for creativity with low raymarch count volumetrics.
Many looks can be found with a few adjustments but the low ray march count makes it tricky and interesting.
One of the challenges is creating depth with the lower raymarch count. 
Accumulated glow/HDR is reduced. Less steps are taken along z and the distance along z is reduced.
Ways I added percieved depth here is through movement, turbulence, 
color contrast of background, volumetric glow, and color variation along length(p.xy).

Notes: 
Aces tonemap > Tanh for higher contrast look in this scene.
Dot noise - volume that we are moving through.
Clear Tunnel for "Camera": 2.-length(p.xy)
Turbulent Effect: s+=abs(sin(p.z)); - Movement along z creates turbulent effect
Color like "Luminence": 1.+sin(i))/s; - s gives glow and sin(i) gives iterative color for each raymarch 
Volumetrics: p+=d*s; Accumulate along each step of d
*/

// NOTE: Assuming iMusicSync0 and iMusicSync1 are passed as uniforms in your environment.
// uniform float iMusicSync0;
// uniform float iMusicSync1;

// 1D Random function (your zoom code used it, so I am providing it here)
float rand(float n) { 
    return fract(sin(n) * 43758.5453123); 
}

// 2d rotation matrix
vec2 r(vec2 v, float t) {
    float s = sin(t), c = cos(t);
    return mat2(c, -s, s, c) * v;
}

// ACES tonemap
vec3 a(vec3 c) {
    mat3 m1 = mat3(0.59719, 0.07600, 0.02840, 0.35458, 0.90834, 0.13383, 0.04823, 0.01566, 0.83777);
    mat3 m2 = mat3(1.60475, -0.10208, -0.00327, -0.53108, 1.10813, -0.07276, -0.07367, -0.00605, 1.07602);
    vec3 v = m1 * c, a = v * (v + 0.0245786) - 0.000090537, b = v * (0.983729 * v + 0.4329510) + 0.238081;
    return m2 * (a / b);
}

// Xor's Dot Noise
float n(vec3 p) {
    const float PHI = 1.618033988;
    const mat3 GOLD = mat3(
        -0.571464913, +0.814921382, +0.096597072,
        -0.278044873, -0.303026659, +0.911518454,
        +0.772087367, +0.494042493, +0.399753815);
    return dot(cos(GOLD * p), sin(PHI * p * GOLD));
}

// =========================================================================
// ENCAPSULATED VORTEX LOGIC
// Instead of mainImage, it takes in coordinates 'u' and returns a vec4 color
// =========================================================================
vec4 getVortexColor(vec2 u) {
    vec4 o = vec4(0.0);
    
    // Safety check: Explicitly initialized i, p, and l here. 
    // The original code left them undefined, which works on some GPUs (NVIDIA) 
    // but results in a black screen on others (Mac/M1/WebGL).
    float i = 0.0, s, t = iTime;
    vec3 p = vec3(0.0, 0.0, t), l = vec3(0.0), b, d;
    
    d = normalize(vec3(2.0 * u - iResolution.xy, iResolution.y));
    
    for (; i < 10.0; i++) {
        b = p;
        b.xy = r(sin(b.xy), t * 1.75 + b.z * 3.5);
        s = 0.002 + abs(n(b * 12.0) / 12.0 - n(b)) * 0.4;
        s = max(s, 2.0 - length(p.xy));
        s += abs(p.y * 0.75 + sin(p.z + t * 0.1 + p.x * 1.5)) * 0.2;
        p += d * s;
        l += (1.0 + sin(i + length(p.xy * 0.1) + vec3(3.0, 1.5, 1.0))) / s;
    }
    
    o.rgb = a(l * l / 600.0);
    o.a = 1.0;
    return o;
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    // Divide the screen coordinates by the resolution to get 0.0 - 1.0
    vec2 uv = fragCoord / iResolution.xy;   
    vec2 centered_uv = uv - 0.5;
    vec2 final_uv = centered_uv + 0.5;
    vec2 transformedCoord = final_uv * iResolution.xy;

    // Calculate the vortex at these transformed pixel coordinates
    vec4 resultColor = getVortexColor(transformedCoord);
    fragColor = resultColor;
}
