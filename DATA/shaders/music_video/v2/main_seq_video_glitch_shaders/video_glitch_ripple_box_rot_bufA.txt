float cubicInOut(float t) {
    return t < 0.5 ? 4.0 * t * t * t : 0.5 * pow(2.0 * t - 2.0, 3.0) + 1.0;
}

float getTime(float t, float duration) {
    return clamp(t, 0.0, duration) / duration;
}

float getRotAnimationValue(float timeInput) {
    float frame = mod(timeInput, 16.0);
    float time = frame;
    float val = 0.0;
    float duration = 1.2;
    
    if (frame >= 1.0 && frame < 5.0) {
        time = getTime(time - 1.0, duration);
        val = cubicInOut(time) * 90.0;
    } else if (frame >= 5.0 && frame < 9.0) {
        time = getTime(time - 5.0, duration);
        val = 90.0 + cubicInOut(time) * 90.0;
    } else if (frame >= 9.0 && frame < 13.0) {
        time = getTime(time - 9.0, duration);
        val = 180.0 + cubicInOut(time) * 90.0;
    } else if (frame >= 13.0) {
        time = getTime(time - 13.0, duration);
        val = 270.0 + cubicInOut(time) * 90.0;
    }
    return val;
}

// HOW TO CHOOSE DIRECTION:
// This function replaces the basic 2D 'Rot()' from Shader 1.
// It allows rotation around ANY arbitrary 3D axis.
mat3 rotate3D(vec3 axis, float angle) {
    axis = normalize(axis);
    float s = sin(angle);
    float c = cos(angle);
    float oc = 1.0 - c;
    
    return mat3(
        oc * axis.x * axis.x + c,           oc * axis.x * axis.y - axis.z * s,  oc * axis.z * axis.x + axis.y * s,
        oc * axis.x * axis.y + axis.z * s,  oc * axis.y * axis.y + c,           oc * axis.y * axis.z - axis.x * s,
        oc * axis.z * axis.x - axis.y * s,  oc * axis.y * axis.z + axis.x * s,  oc * axis.z * axis.z + c
    );
}

float EffectDuration = 2.0 + iMusicSync2*2;
float EffectFadeInTimeFactor = 0.5;
float EffectWidth = 0.6 + iMusicSync2*2;
float EffectMaxTexelOffset = 40.0 + iMusicSync2*3;

vec2 GetOffsetFromCenter(vec2 screenCoords, vec2 screenSize) {
    vec2 halfScreenSize = screenSize / 2.0;
    return (screenCoords.xy - halfScreenSize) / min(halfScreenSize.x, halfScreenSize.y);
}

vec2 GetDistortionTexelOffset(vec2 offsetDirection, float offsetDistance, float time) {
    float progress = mod(time, EffectDuration) / EffectDuration;
    float halfWidth = EffectWidth / 2.0;
    float lower = 1.0 - smoothstep(progress - halfWidth, progress, offsetDistance);
    float upper = smoothstep(progress, progress + halfWidth, offsetDistance);
    
    float band = 1.0 - (upper + lower);
    float strength = 1.0 - progress;
    float fadeStrength = smoothstep(0.0, EffectFadeInTimeFactor, progress);
    
    return band * strength * fadeStrength * offsetDirection * EffectMaxTexelOffset;
}

vec3 GetTextureOffset(vec2 coords, vec2 textureSize, vec2 texelOffset) {
    vec2 texelSize = 1.0 / textureSize;
    vec2 offsetCoords = coords + texelSize * texelOffset;
    vec2 halfTexelSize = texelSize / 2.0;
    vec2 clampedOffsetCoords = clamp(offsetCoords, halfTexelSize, 1.0 - halfTexelSize);
    return texture(iChannel0, clampedOffsetCoords).rgb;
}

// Wrapper that applies Shader 2 to our projected 3D coordinates
vec3 ApplyGlitchRipple(vec2 uv, float time, vec2 screenSize) {
    // Convert 0..1 UVs into virtual screen coordinates for the ripple math
    vec2 virtualCoords = uv * screenSize;
    
    vec2 offsetFromCenter = GetOffsetFromCenter(virtualCoords, screenSize);
    vec2 offsetDirection = normalize(-offsetFromCenter);
    float offsetDistance = length(offsetFromCenter);
    
    vec2 offset = GetDistortionTexelOffset(offsetDirection, offsetDistance, time);
    
    // Standard flip for Shadertoy textures
    uv.y = 1.0 - uv.y;
    
    return GetTextureOffset(uv, screenSize, offset);
}

#define MAX_STEPS 100
#define MAX_DIST 10.0
#define SURF_DIST 0.001

// The dimensions of our rotating "image box" (width, height, depth)
const vec3 BOX_SIZE = vec3(1.35, 0.55, 1.35);

float sdBox(vec3 p, vec3 b) {
    vec3 q = abs(p) - b;
    return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
}

vec2 GetDist(vec3 p) {
    // Get animation angle and apply custom 3D axis rotation
    float angle = radians(getRotAnimationValue(iTime));
    p = p * rotate3D(ROTATION_AXIS, -angle);
    
    float d = sdBox(p, BOX_SIZE);
    return vec2(d, 1.0);
}

vec3 GetNormal(vec3 p) {
    vec2 e = vec2(0.001, 0);
    vec3 n = GetDist(p).x - vec3(
        GetDist(p - e.xyy).x,
        GetDist(p - e.yxy).x,
        GetDist(p - e.yyx).x
    );
    return normalize(n);
}

// --- REUSABLE WAVE FUNCTION ---
// fragCoord : Current pixel coordinates
// amplitude : Height of the wave
// yCenter   : The Y axis offset in pixels
// baseColor : The RGB color of the wave
// freq      : How tight the sine wave is (0.02 from your original code)
// phaseShift: Used with time to make the waves scroll horizontally

vec3 drawAudioWave(vec2 fragCoord, float amplitude, float yCenter, vec3 baseColor, float freq, float phaseShift) {
        
    // Graceful fallback from your Pygame code: 
    // If amplitude is tiny, transition smoothly to a white line instead of snapping harshly.
    vec3 waveColor = mix(vec3(1.0, 1.0, 1.0), baseColor, smoothstep(0.0, 2.0, amplitude));

    // Calculate the mathematical Y position of the sine wave for this X pixel
    float waveY = yCenter + amplitude * sin(fragCoord.x * freq + phaseShift);
    
    // Vertical distance from the current pixel to the wave
    float dist = abs(fragCoord.y - waveY);
    
    // --- COOL FACTOR: THE GLOW ---
    // Core thickness creates a bright center, exponential falloff creates a soft neon glow
    float coreThickness = 0.0;

    if (amplitude < 2.0) {
        coreThickness = 0.1;
    }
    else {
        coreThickness = 0.5;
    }

    float glowSpread = 3.0;
    float intensity = (coreThickness / (dist + 1e-4)) + exp(-dist / glowSpread);
    
    // --- COOL FACTOR: SMOOTH EDGES ---
    // Instead of hard cuts at 450 and (Width - 400), we fade the line in and out beautifully
    float minX = 450.0;
    float maxX = iResolution.x - 400.0; 
    float mask = smoothstep(minX - 50.0, minX, fragCoord.x) * 
                 (1.0 - smoothstep(maxX, maxX + 50.0, fragCoord.x));
                 
    return waveColor * intensity * mask;
}


vec3 B2_spline(vec3 x) { // returns 3 B-spline functions of degree 2
    vec3 t = 3.0 * x;
    vec3 b0 = step(0.0, t)     * step(0.0, 1.0-t);
	vec3 b1 = step(0.0, t-1.0) * step(0.0, 2.0-t);
	vec3 b2 = step(0.0, t-2.0) * step(0.0, 3.0-t);
	return 0.5 * (
    	b0 * pow(t, vec3(2.0)) +
    	b1 * (-2.0*pow(t, vec3(2.0)) + 6.0*t - 3.0) + 
    	b2 * pow(3.0-t,vec3(2.0))
    );
}

float bump(float x) {
	return abs(x) > 1.0 ? 0.0 : 1.0 - x * x;
}

}

        vec3 p = ro + rd * dO;
        vec2 dS = GetDist(p);

        // Reverse the rotation so we can map the 2D texture flatly onto the box faces

        mat3 invRot = rotate3D(ROTATION_AXIS, angle);
        vec3 localP = invRot * p;
        vec3 localN = invRot * n;
        
        // Map 3D Hit position to 2D UV Coordinates (0.0 to 1.0) based on which face is hit
        vec2 faceUV = vec2(0.0);
        if (abs(localN.z) > 0.5) {
            faceUV = localP.xy / BOX_SIZE.xy;
        } else if (abs(localN.x) > 0.5) {
            faceUV = vec2(localP.z, localP.y) / vec2(BOX_SIZE.z, BOX_SIZE.y);
        } else {
            faceUV = localP.xz / vec2(BOX_SIZE.x, BOX_SIZE.z);
