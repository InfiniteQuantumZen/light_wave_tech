void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
	float t=iTime*.1;
	vec2 uv = gl_FragCoord.xy / iResolution.xy-.5;
	vec2 ouv=uv;
	uv.x*=iResolution.x/iResolution.y;
	vec3 rd=normalize(vec3(uv,2.));
	rd.xy*=mat2(cos(t),sin(t),-sin(t),cos(t));
	vec3 ro=vec3(t+sin(t*6.53583)*.05, .01+sin(t*352.4855)*.0015, -t*(3.0+(iMusicSync0/10)));
	vec3 p=ro+(iMusicSync0/10);
	float v=0., td=-mod(ro.z,.005);
	for (int r=0; r<150; r++) {
		v+=pow(max(0., .01-length(abs(.01-mod(p,.02))))/.01, 10.)*exp(-2.*pow((1.+td),2.));
		p=ro+rd*td;
		td+=.005;
	}
vec3 myColor = vec3(0.1+iMusicSync0+iMusicSync1, 0.1+iMusicSync0+iMusicSync2, 0.5+iMusicSync0); // Pick any RGB values between 0.0 and 1.0
fragColor = vec4(vec3(v) * myColor, 0.)*(8.0+(iMusicSync0/10))*max(0.,1.-length(ouv*ouv)*2.5);
}
