// --- REUSABLE WAVE FUNCTION ---
// fragCoord : Current pixel coordinates
// amplitude : Height of the wave
// yCenter   : The Y axis offset in pixels
// baseColor : The RGB color of the wave
// freq      : How tight the sine wave is (0.02 from your original code)
// phaseShift: Used with time to make the waves scroll horizontally

vec3 drawAudioWave(vec2 fragCoord, float amplitude, float yCenter, vec3 baseColor, float freq, float phaseShift) {
        
    // Graceful fallback from your Pygame code: 
    // If amplitude is tiny, transition smoothly to a white line instead of snapping harshly.
    vec3 waveColor = mix(vec3(1.0, 1.0, 1.0), baseColor, smoothstep(0.0, 2.0, amplitude));

    // Calculate the mathematical Y position of the sine wave for this X pixel
    float waveY = yCenter + amplitude * sin(fragCoord.x * freq + phaseShift);
    
    // Vertical distance from the current pixel to the wave
    float dist = abs(fragCoord.y - waveY);
    
    // --- COOL FACTOR: THE GLOW ---
    // Core thickness creates a bright center, exponential falloff creates a soft neon glow
    float coreThickness = 0.0;

    if (amplitude < 2.0) {
        coreThickness = 0.1;
    }
    else {
        coreThickness = 0.5;
    }

    float glowSpread = 3.0;
    float intensity = (coreThickness / (dist + 1e-4)) + exp(-dist / glowSpread);
    
    // --- COOL FACTOR: SMOOTH EDGES ---
    // Instead of hard cuts at 450 and (Width - 400), we fade the line in and out beautifully
    float minX = 450.0;
    float maxX = iResolution.x - 400.0; 
    float mask = smoothstep(minX - 50.0, minX, fragCoord.x) * 
                 (1.0 - smoothstep(maxX, maxX + 50.0, fragCoord.x));
                 
    return waveColor * intensity * mask;
}


vec3 B2_spline(vec3 x) { // returns 3 B-spline functions of degree 2
    vec3 t = 3.0 * x;
    vec3 b0 = step(0.0, t)     * step(0.0, 1.0-t);
	vec3 b1 = step(0.0, t-1.0) * step(0.0, 2.0-t);
	vec3 b2 = step(0.0, t-2.0) * step(0.0, 3.0-t);
	return 0.5 * (
    	b0 * pow(t, vec3(2.0)) +
    	b1 * (-2.0*pow(t, vec3(2.0)) + 6.0*t - 3.0) + 
    	b2 * pow(3.0-t,vec3(2.0))
    );
}

float bump(float x) {
	return abs(x) > 1.0 ? 0.0 : 1.0 - x * x;
}


// 1. Put this function ABOVE your mainImage function
float rand(float seed) {
    // This is the classic GLSL 1D random generator
    return fract(sin(seed) * 43758.5453123);
}

void mainImage(out vec4 fragColor, vec2 fragCoord) {
    // =========================================================================
    // PART 1: EXACT ROTOZOOM EMULATION WITH DYNAMIC PIVOTS
    // =========================================================================

    // Divide the screen coordinates by the TEXTURE'S resolution
    vec2 uv = fragCoord / iResolution.xy;
    uv.y = 1.0 - uv.y;
    
    // Convert UV to exact screen pixels (This prevents Ultrawide distortion)
    vec2 px = uv * iResolution.xy;

    // Apply the translation shake offset
    vec2 local_px = px - u_pixel_offset;

    // Prepare rotation math
    float rad = radians(u_angle_deg);
    float c = cos(rad);
    float s = sin(rad);

    // Calculate Pygame's expanded Bounding Box! 
    float bbox_w = u_target_size.x * abs(c) + u_target_size.y * abs(s);
    float bbox_h = u_target_size.x * abs(s) + u_target_size.y * abs(c);

    // --> THE SECRET SAUCE MULTIPLIER <--
    vec2 bbox_pivot = vec2(bbox_w, bbox_h) * u_pivot;

    // Shift relative to the chosen pivot
    vec2 centered_px = local_px - bbox_pivot;

    // Apply the exact Inverse Rotation Matrix that matches your orientation
    mat2 rotMat = mat2(c, -s, 
                       s,  c);
    vec2 rotated_px = rotMat * centered_px;

    // Shift back from the pivot to the original unrotated surface coordinates
    vec2 surf_pivot = u_target_size * u_pivot;
    vec2 final_px = rotated_px + surf_pivot;

    // Normalize back to 0.0 - 1.0 UV space using the ZOOMED pixel dimensions
    vec2 final_uv = final_px / u_target_size;

    uv = final_uv;

    // OPTIMIZATION: Cut off the rendering BEFORE the heavy loop if outside the box.
    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
        fragColor = vec4(0.0, 0.0, 0.0, 0.0); // Output transparent black
        return;
    }

    // =========================================================================
    // PART 2: ASCII ART TRAIL GENERATION (Replacing standard texture sample)
    // =========================================================================
    
    vec2 R = iResolution.xy;
    vec2 S = vec2(8.0, 16.0);
    
    // Convert the transformed UV back into pseudo-screen coordinates.
    // This makes the ASCII effect map onto the rotated/zoomed surface!
    vec2 baseU = final_uv * R; 
    
    vec4 asciiColor = vec4(0.0);
    vec4 colorMult = vec4(0.4, 1.0, 0.2, 1.0);
    vec4 fadeColor = vec4(0.9, 0.8, 0.9, 0.0);
    vec4 currentFade = vec4(1.0);
    
    // Look back 30 frames into the past
    for(int i = 0; i < 30; i++) {
        // Calculate the position this pixel was at "i" frames ago, 
        // applying the spatial offset to the transformed coordinates.
        vec2 pastU = baseU + float(i) * vec2(0.5, -2.0);
        
        // Check iChannel0 (Source Image/Video) at that past position
        int charIdx = 140 + int[](68,82,63,69,0,1,54,37,51,21,-27,36)
                          [int(12.0 * texture( iChannel0, ceil(pastU/S)*S/R ).r)];
                          
        // Get the character value from iChannel1 (Font Texture)
        vec4 charColor = texture( iChannel1, ( mod(pastU,S) + vec2(4.0,0.0) )/256.0
                                 + vec2(float(charIdx), float(charIdx/16)) / 16.0 ).r * colorMult;
                                 
        // Accumulate using the fading formula
        asciiColor += (1.0 - asciiColor) * currentFade * charColor;
        
        // Advance the fade multiplier for the next iteration step further back
        currentFade *= fadeColor;
    }

    // Assign the finished ASCII generation to fragColor
    fragColor = asciiColor;

    // =========================================================================
    // PART 3: MUSIC SYNC & FLASH FX
    // =========================================================================

    // MUSIC_VIDEO__MOD
    float colorTimer = floor(iTime * 2.0); 

    vec3 flashColor = vec3(
        rand(colorTimer),           
        rand(colorTimer + 10.0),    
        rand(colorTimer + 20.0)     
    );

    // Add the flash to the transformed screen
    vec3 finalColor = fragColor.rgb + (flashColor * iMusicSync2 * 1.2);

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 

    // =========================================================================
    // PART 4: WAVE FUNCTION
    // =========================================================================

    // Define your colors (Normalized from 0-255 to 0.0-1.0)
    vec3 MAGENTA = vec3(250.0, 34.0, 197.0) / 255.0;
    vec3 GREEN   = vec3(8.0, 240.0, 124.0) / 255.0;
    vec3 CYAN    = vec3(8.0, 224.0, 240.0) / 255.0;
    vec3 GOLD    = vec3(240.0, 224.0, 8.0) / 255.0;
    vec3 PURPLE  = vec3(187.0, 34.0, 250.0) / 255.0;
    vec3 WHITE   = vec3(1.0, 1.0, 1.0);
    
    // Calculate accurate Y offsets accounting for OpenGL's bottom-up coordinate system
    //float yTop    = iResolution.y - 60.0;   // 60 pixels from the top
    //float yBottom = iResolution.y - 1000.0; // 1410 pixels from the top (30 from bottom on 1440p)
    //float yBottom = iResolution.y - 1410.0; // 1410 pixels from the top (30 from bottom on 1440p)

    float yTop    = 1350.0;   // 60 pixels from the top
    float yBottom = 140.0; // 1410 pixels from the top (30 from bottom on 1440p) 

    finalColor = vec3(0.0);
    float freq = 0.02; // From your python code
    
    // --- COOL FACTOR: TIME SCROLLING ---
    // Multiplying time creates an offset so the waves visibly "flow" across the screen
    float speed = iTime * 5.0;

    // Additive blending: Adding the colors together means where they overlap, they blend brightly to white
    //finalColor += drawAudioWave(fragCoord, u_amp_top.x, yTop, MAGENTA, freq, speed * 1.5);
    //finalColor += drawAudioWave(fragCoord, u_amp_top.y, yTop, GREEN,   freq, speed * 1.2);
    //finalColor += drawAudioWave(fragCoord, u_amp_top.z, yTop, CYAN,    freq, speed * 1.0);
    
    // --- BOTTOM WAVES ---
    // Use negative speed to make bottom waves flow in the opposite direction for contrast!
    finalColor += drawAudioWave(fragCoord, u_amp_bottom.x*2.0, yBottom, GOLD,   freq, speed * -3.0);
    finalColor += drawAudioWave(fragCoord, u_amp_bottom.y*2.0, yBottom, PURPLE, freq, speed * -3.2);
    finalColor += drawAudioWave(fragCoord, u_amp_bottom.z*2.0, yBottom, WHITE,  freq, speed * -3.5);

    // Optional: Tone-mapping to prevent the glowing hot spots from burning out entirely
    finalColor = finalColor / (1.0 + finalColor * 0.2); 

    // Add the wave to the screen
    finalColor = fragColor.rgb + finalColor;

    // Output the final mixed color
    fragColor = vec4(finalColor, fragColor.a); 





    // =========================================================================
    // PART 5: AUDIO SPECTRUM 1
    // =========================================================================

    // Get the standard normalized coordinates (0.0 to 1.0)
    uv = (fragCoord.xy / iResolution.xy);

    // --- START OF SCALING & POSITIONING ---
    
    // Set exactly where you want the CENTER of the visualizer (in pixels).
    // X = middle of screen. Y = 100 pixels up from the bottom.
    vec2 TARGET_POS = vec2(iResolution.x * 0.5, 280.0); 
    
    // Set exactly how large you want the visualizer to be (in pixels).
    // X = full width of screen. Y = restricted to 100 pixels tall.
    vec2 TARGET_SIZE = vec2((iResolution.x), 700.0);
    
    // Remap the UVs based on our targets
    uv = (uv - (TARGET_POS / iResolution.xy)) * (iResolution.xy / TARGET_SIZE) + 0.5;

    // --- END OF SCALING & POSITIONING ---
    // (The rest of the code is unchanged, but uses our newly squished 'uv')

    float fVBars = 40.0;
    float fHSpacing = 1.00;
    
    float x = floor(uv.x * fVBars)/fVBars;
    float fSample = texture(iChannel3, vec2(abs(2.0 * x - 1.0), 0.25)).x;
  
    float squarewave = sign(mod(uv.x, 1.0/fVBars)-0.004);
    float fft = squarewave * fSample* 0.5; 
    
    float fHBars = 0.0; //100.0;
    float fVSpacing = 0.0; //0.180;
    float fVFreq = (uv.y * 3.14);
    fVFreq = sign(sin(fVFreq * fHBars)+1.0-fVSpacing);

    vec2 centered = vec2(1.0) * uv - vec2(1.0);
    float t = iTime / 100.0;
    float polychrome = 1.0;
    vec3 spline_args = fract(vec3(polychrome*uv.x-t) + vec3(0.0, -1.0/3.0, -2.0/3.0));
    vec3 spline = B2_spline(spline_args);
    
    float f = abs(centered.y);
    vec3 base_color  = vec3(0.95, 0.95, 0.95) - f*spline;
    vec3 flame_color = pow(base_color, vec3(3.0));
    
    float tt = 0.3 - uv.y;
    float df = sign(tt);
    df = 0.0; //(df + 1.0)/0.5;
    vec3 col = flame_color * vec3(1.0 - step(fft, abs(0.3-uv.y))) * vec3(fVFreq);
    col -= col * df * 0.0180;

    // Add the audio viz to the screen
    finalColor = fragColor.rgb + col;

    // Output the final mixed color
    //fragColor = vec4(finalColor, fragColor.a);
    fragColor = vec4(finalColor, 0.0);


    // =========================================================================
    // PART 6: AUDIO SPECTRUM 2
    // =========================================================================

    // Set exactly where you want the CENTER of the visualizer (in pixels).
    // X = middle of screen. Y = 100 pixels up from the bottom.
    TARGET_POS = vec2(iResolution.x * 0.5, 430.0); 
    
    // Set exactly how large you want the visualizer to be (in pixels).
    // X = full width of screen. Y = restricted to 100 pixels tall.
    TARGET_SIZE = vec2(iResolution.x, 2400.0);
    
    // Remap the UVs based on our targets
    uv = (uv - (TARGET_POS / iResolution.xy)) * (iResolution.xy / TARGET_SIZE) + 0.5;

    
    // Because the math in this shader stretches lines to infinity, 
    // we need to "cut off" the rendering if the pixels are outside our 100px box.
    //if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
    //    fragColor = vec4(0.0, 0.0, 0.0, 0.0); // Output transparent black
    //    return;
    //}


    c = 4.0;
    vec3 color = vec3(1.0);
    color.z = bump(c * (uv.x - 0.75));
    color.y = bump(c * (uv.x - 0.5));
    color.x = bump(c * (uv.x - 0.25));
	
    float line = abs(0.02 / abs(0.5-uv.y) );
    uv.y = abs( uv.y - 0.5 );
	
    vec4 soundWave = texture( iChannel3, vec2(abs(0.5-uv.x)+0.005, uv.y) ).rrrr;
    color *= line * (1.0 - 2.0 * abs( 0.5 - uv.xxx ) + pow( soundWave.y, 10.0 ) * 50.0 );
	
    fragColor = fragColor + vec4(color, 0.0); 

}
