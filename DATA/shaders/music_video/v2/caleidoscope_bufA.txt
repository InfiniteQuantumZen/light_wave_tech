// Функция для создания яркой радуги (Hue to RGB)
vec3 hsb2rgb(vec3 c) {
    vec3 rgb = clamp(abs(mod(c.x*6.0+vec3(0.0,4.0,2.0),6.0)-3.0)-1.0, 0.0, 1.0);
    rgb = rgb*rgb*(3.0-2.0*rgb); // Сглаживание
    return c.z * mix(vec3(1.0), rgb, c.y);
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = (fragCoord * 2.0 - iResolution.xy) / iResolution.y;
    
    // Вращение
    float angle = iTime * 0.1;
    mat2 rot = mat2(cos(angle), -sin(angle), sin(angle), cos(angle));
    uv *= rot;

    // Жестко 6 граней
    float a = atan(uv.y, uv.x);
    float r = length(uv);
    float sides = 6.0;
    a = mod(a, 2.0 * 3.14159 / sides);
    a = abs(a - 3.14159 / sides);
    uv = vec2(cos(a), sin(a)) * r;

    vec2 uv0 = uv;
    vec3 finalColor = vec3(0.0);
    
    for (float i = 0.0; i < 4.0; i++) {
        uv = fract(uv * 1.5) - 0.5;

        float d = length(uv) * exp(-length(uv0));

        // Постепенный шифт HUE (радуги)
        // iTime * 0.2 — скорость смены цветов
        // length(uv0) — распределение цвета от центра
        float hue = fract(iTime * 0.15 + length(uv0) * 0.5 + i * 0.1);
        vec3 col = hsb2rgb(vec3(hue, 0.8, 1.0));

        d = sin(d * 8.0 + iTime) / 8.0;
        d = abs(d);
        d = pow(0.01 / d, 1.2);

        finalColor += col * d;
    }
        
    fragColor = vec4(finalColor, 1.0);
}
