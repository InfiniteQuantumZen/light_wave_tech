void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
   
    float loopTime = mod(iTime, 30.0);
    
    vec2 uv = fragCoord/iResolution.xy;
    vec2 q = uv - vec2(0.5, 0.5);
    vec2 a = uv + mod(0.5, 0.5);     
    vec3 col = vec3(0.5, 0.7, 1.0);
    
    
    float r = 0.02 + 0.9 * sin(atan(q.y, q.x) * loopTime * 5.0); 
    float d = 0.2 * 6.0 * fract(atan(a.y, a.x) * loopTime * a.x * 2.0);
    
    col *= mix(r, r*0.0001, length(q / d*3.23)); 
    fragColor = vec4(col, 1.0); 
}
