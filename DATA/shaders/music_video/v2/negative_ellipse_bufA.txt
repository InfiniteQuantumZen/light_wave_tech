float F = 0.2;

float sinT(in float f){
    return sin(mod(F*f*iTime,6.28318));
}
float cosT(in float f){
    return cos(mod(F*f*iTime,6.28318));
}

float totalDist(in vec2 p){
    float m = 0.0;

    m +=  1.0 * length(p - vec2(-2.0*cosT(0.4298),-2.0*sinT(0.943)));
    m += -1.0 * length(p - vec2(2.0*sinT(2.134),-2.0*cosT(-4.3578)));
    m +=  1.0 * length(p - vec2(-2.0*sinT(0.9),-2.0*cosT(0.824)));
    m += -1.0 * length(p - vec2(2.0*sinT(1.124),-2.0*cosT(-2.12384)));
    
    return m;
}

//https://iquilezles.org/articles/palettes
vec3 pal( in float t, in vec3 a, in vec3 b, in vec3 c, in vec3 d )
{
    return a+b*cos(6.28318*(c*t+d));
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = 4.0*(fragCoord - 0.5*iResolution.xy)/min(iResolution.x,iResolution.y);
    
    vec3 col = vec3(0.0);

    float d = totalDist(uv);

    col = pal(0.25 + d, vec3(0.5,0.5,0.5),vec3(0.5,0.5,0.5),vec3(1.0,1.0,1.0),vec3(0.3,0.20,0.20));
    
    //red line
    /*
    if(abs(d) <= 0.05){
        col = vec3(1.0,0.0,0.0);
    }
    */
    
    fragColor = vec4(col,1.0);
}
